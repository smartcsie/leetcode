    /**
 * 題目：1239. Maximum Length of a Concatenated String with Unique Characters
 * 難度：Medium
 * 描述：給定字串陣列 arr，從中選出若干個字串（按順序串接），
 *       使串接後的字串每個字母都只出現一次，回傳可得的最大長度。
 *
 * 分類主題：backtracking-combination
 * 技巧：字串轉成 26 位元遮罩, 遮罩重疊就剪枝, 用 popcount(used) 當目前長度, 自己有重複字母的字預先丟掉
 *
 * 時間複雜度：O(2^n)，n 為篩選後可用的字串數；遇到重疊就整棵子樹跳過，實際通常遠小於此
 * 空間複雜度：O(n)，遞迴深度與遮罩陣列
 *
 * 解法思路：DFS 選或不選 + 重疊剪枝
 * 1. 每個字串轉成 26 位元的遮罩，第 k 位為 1 代表含有第 k 個字母。
 *    字串自己就有重複字母（轉換過程中某一位已經是 1）者不可能被選，直接丟掉。
 * 2. dfs(start, used)：used 是目前已選字串的字母聯集。
 *    因為所有已選字母都互不相同，所以目前串接長度就是 popcount(used)。
 * 3. 從 start 開始往後挑下一個字串，若它的遮罩與 used 有重疊（&），
 *    串接後會有重複字母，這個字串不能選，直接跳過。
 * 4. 可以選就遞迴到 i + 1，used 更新為 used | masks[i]。
 *    子集內的字串依索引遞增，同一組合不會以不同順序重複出現。
 * 5. 每個節點本身就是一個合法組合，所以進入函式時就先用 popcount(used) 當候選答案，
 *    再與子樹中更長的結果比較。
 * 6. 衝突時直接跳過，比列舉全部 2^n 個子集再檢查更有效率：
 *    包含衝突組合的所有情況都不會被走到。
 */
class Solution {
public:
    int maxLength(vector<string>& arr) {
        vector<int> masks;  // 每個可用字串的字母遮罩
        for (const string& w : arr) {
            int m = 0;
            bool ok = true;
            for (char c : w) {
                int bit = 1 << (c - 'a');
                if (m & bit) { ok = false; break; }  // 字自己有重複字母，整個丟掉
                m |= bit;
            }
            if (ok) masks.push_back(m);
        }
        return dfs(masks, 0, 0);
    }

private:
    // start：這一層只能從 masks[start] 之後挑字串；used：目前已選字串的字母聯集
    int dfs(const vector<int>& masks, int start, int used) {
        int best = __builtin_popcount(used);  // 目前用到的不同字母數，就是目前長度
        for (int i = start; i < masks.size(); i++) {
            if (masks[i] & used) continue;    // 字母重疊，這條路不用走
            best = max(best, dfs(masks, i + 1, used | masks[i]));
        }
        return best;
    }
};

    /**
 * 題目：2397. Maximum Rows Covered by Columns
 * 難度：Medium
 * 描述：給定 m x n 的 0/1 矩陣 matrix 與整數 numSelect。
 *       從中選出恰好 numSelect 欄，若某一列所有為 1 的格子所在的欄都被選中，
 *       該列就算「被覆蓋」（全為 0 的列一定被覆蓋）。
 *       回傳最多能覆蓋多少列。
 *
 * 分類主題：backtracking-combination
 * 技巧：每一列轉成欄遮罩, (r & chosen) == r 判斷列是否被覆蓋, DFS 選恰好 k 欄, i + left <= n 剪枝
 *
 * 時間複雜度：O(C(n, k) * m)，k 為 numSelect，每種選法要檢查 m 列
 * 空間複雜度：O(m + n)，rows 陣列與遞迴深度
 *
 * 解法思路：DFS 列舉欄的組合
 * 1. 每一列轉成一個遮罩 rows[i]：第 j 位為 1 代表該列在第 j 欄有 1。
 * 2. 一列被覆蓋，當且僅當它所有的 1 都落在已選的欄裡，
 *    也就是 (rows[i] & chosen) == rows[i]。
 *    全為 0 的列 rows[i] = 0，永遠成立，符合題意。
 * 3. 用 DFS 從 n 欄中挑恰好 numSelect 欄：
 *    - left：還要再選幾欄；start：只能從第 start 欄之後挑，避免同一組合重複；
 *      chosen：目前已選欄的遮罩。
 *    - left 變成 0 時選夠了，掃過所有列統計被覆蓋的列數並回傳。
 * 4. 剪枝 i + left <= n：選了第 i 欄之後，後面剩下的欄數必須還能湊滿 left - 1 欄，
 *    不夠的話這個分支不可能選滿，不用走。
 * 5. 與 1239 的差別：這題沒有「選了會衝突」的限制，任何欄都能選，
 *    改成用 left 控制「恰好選幾個」。
 */
class Solution {
public:
    int maximumRows(vector<vector<int>>& matrix, int numSelect) {
        int m = matrix.size(), n = matrix[0].size();
        vector<int> rows(m, 0);  // rows[i]：第 i 列的 1 所在的欄遮罩
        for (int i = 0; i < m; i++)
            for (int j = 0; j < n; j++)
                if (matrix[i][j]) rows[i] |= 1 << j;
        return dfs(rows, n, numSelect, 0, 0);
    }

private:
    // left：還要再選幾欄；start：只能從第 start 欄之後挑；chosen：已選欄的遮罩
    int dfs(const vector<int>& rows, int n, int left, int start, int chosen) {
        if (left == 0) {
            int covered = 0;
            for (int r : rows)
                if ((r & chosen) == r) covered++;  // 這列的 1 全部都在已選欄裡
            return covered;
        }

        int best = 0;
        // i + left <= n：選了第 i 欄後，後面剩下的欄數仍足夠湊滿 left - 1 欄
        for (int i = start; i + left <= n; i++)
            best = max(best, dfs(rows, n, left - 1, i + 1, chosen | (1 << i)));
        return best;
    }
};

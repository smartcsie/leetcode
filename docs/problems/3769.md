```cpp
/**
 * 題目：3769. Sort Integers by Binary Reflection
 * 難度：Easy
 * 描述：正整數的「二進位反射」是把它的二進位表示（不含前導 0）反轉後，
 *       再當成二進位數字讀回來的值。例如 6 = 110 → 011 → 3。
 *       依反射值由小到大排序 nums，反射值相同時，原數字較小的在前。
 *
 * 分類主題：sorting-custom-comparator
 * 技巧：逐位反轉 r = (r << 1) | (y & 1), pair 排序同時處理主鍵與次鍵, 先算 key 再排序
 *
 * 時間複雜度：O(N log N + N log M)，M 為 nums 中的最大值（每個數反轉約 log M 位）
 * 空間複雜度：O(N)
 *
 * 解法思路：計算排序鍵 + pair 排序
 * 1. 對每個數字 y，從最低位開始逐位取出，接到結果 r 的尾端：
 *    - y & 1 取出目前最低位
 *    - r << 1 騰出一位，再用 | 補上剛取出的位
 *    - y >>= 1 丟掉已處理的最低位
 *    因為 y 的前導 0 本來就不存在，迴圈在 y 變 0 時自然停止，
 *    不會把前導 0 反轉成尾端的 0。
 * 2. 把 (反射值, 原數字) 存成 pair。pair 預設先比 first、再比 second，
 *    剛好符合「反射值由小到大，相同時原數字由小到大」，不需要自訂比較器。
 * 3. 排序後依序取出 second，就是答案。
 * 4. 溢位檢查：nums[i] <= 10^9 < 2^30，反射值最多 30 位，
 *    不超過 2^30，int 放得下。
 */

class Solution {

public:
    vector<int> sortByReflection(vector<int>& nums) {
        int n = nums.size();
        vector<pair<int, int>> refToNum;
        for(int i = 0; i < nums.size(); i++) {
            int y = nums[i];
            int r = 0;
            while(y > 0) {
                r = (r << 1) | (y & 1);
                y >>= 1;
            }
            refToNum.push_back({r, nums[i]});
        }
        sort(refToNum.begin(), refToNum.end());
        vector<int> ans(n);
        for(int i = 0; i < n; i++) ans[i] = refToNum[i].second;
        return ans;
    }
};
```

> **排reflect不相等，reflect小的在前面**<br>
> **排序頻率相等，數字小在前面**<br>

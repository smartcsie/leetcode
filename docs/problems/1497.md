/*
 * 題目：1497. Check If Array Pairs Are Divisible by k
 * 難度：Medium
 * 描述：給定長度為偶數的整數陣列 arr 與整數 k，判斷能否把陣列分成 n / 2 對，使每一對的和都能被 k 整除。
 * 分類主題：hash-table-counting
 * 技巧：餘數計數, 負數取餘修正, 餘數互補配對, 餘數 0 與 k / 2 的特例
 * 時間複雜度：O(n + k)
 * 空間複雜度：O(k)
 * 解法思路：
 *   1. 兩數和能被 k 整除 ⇔ 兩數餘數相加等於 0 或 k，所以只需要看每個數除以 k 的餘數
 *   2. C++ 中負數取餘會得到負值，先用 ((x % k) + k) % k 統一轉成 [0, k - 1]
 *   3. 用 counts[r] 統計餘數為 r 的個數
 *   4. 餘數 0 的數只能和餘數 0 的數配對，個數必須是偶數
 *   5. 餘數 i 只能和餘數 k - i 配對，兩邊個數必須相等；只需檢查 1 到 k / 2，每一對檢查一次
 *   6. k 為偶數時，餘數 k / 2 的配對對象是自己，個數必須是偶數
 *   7. 全部通過回傳 true
 */
class Solution {
public:
    bool canArrange(vector<int>& arr, int k) {
        // counts[r] = 餘數為 r 的元素個數 (r 範圍 0 ~ k-1)
        vector<int> counts(k, 0);

        // 統計各餘數的個數
        // C++ 中負數取餘會得到負值,所以先 +k 再 %k,統一轉成 [0, k-1]
        for (int& x : arr) counts[((x % k) + k) % k]++;

        // 餘數 0 的數只能和餘數 0 的數配對 (0 + 0 = 0),所以個數必須是偶數
        if (counts[0] & 1) return false;

        // 只需檢查 1 ~ k/2,因為餘數 i 與 k-i 是一對,每一對只要檢查一次
        for (int i = 1; i <= k / 2; i++) {
            if (i == k - i) {
                // k 為偶數時,餘數 k/2 的配對對象是自己 (k/2 + k/2 = k)
                // 所以個數必須是偶數
                if (counts[i] & 1) return false;
            } else if (counts[i] != counts[k - i]) {
                // 餘數 i 只能和餘數 k-i 配對,兩邊個數必須相等
                return false;
            }
        }

        // 所有餘數都能完美配對
        return true;
    }
};

```cpp
/**
 * 題目：404. Sum of Left Leaves (左葉子節點之和)
 * 難度：簡單 (Easy)
 * 描述：計算二元樹中所有左葉子節點的和。
 *
 * 時間複雜度：O(N)
 * 空間複雜度：O(H)
 *
 * 設計思路：
 * 1. 遞迴檢查每個節點： 1. 若該節點的左子節點是葉子，則直接加上該值。
 * 2. 2. 若不是，則遞迴搜尋左子樹與右子樹。
 */

class Solution {
public:
    int sumOfLeftLeaves(TreeNode* root) {
        if(!root) return 0;
        int sum = 0;
        if(root->left) {
            if(!root->left->left && !root->left->right) sum += root->left->val;
            else sum += sumOfLeftLeaves(root->left);
        }   
        sum += sumOfLeftLeaves(root->right);
        return sum;
    }
};
```

```cpp
// 題目：404. Sum of Left Leaves
// 難度：Easy
// 時間複雜度：O(n)（每條邊最多走兩次）
// 空間複雜度：O(1)（不用遞迴堆疊或佇列）
// 分類主題：tree / binary-tree
// 技巧：Morris Traversal（線索二元樹）：暫時把中序前驅的 right 指回目前節點當作「回頭的路」，走完再拆掉

class Solution {
public:
    int sumOfLeftLeaves(TreeNode* root) {
        int totalSum = 0;
        TreeNode* currentNode = root;
        while(currentNode != nullptr) {
            // 沒有左子樹：不需要回頭，直接往右走
            if(currentNode->left == nullptr) {
                currentNode = currentNode->right;
            } else {
                // previous 先指向左子節點，之後會往右走到左子樹最右邊的節點
                TreeNode* previous = currentNode->left;

                // 左子節點是葉子就計入答案
                // 只會在第一次走到 currentNode 時計入：
                // 第二次回來時，這個葉子的 right 已被接成線索（指向 currentNode），
                // 不再是 nullptr，所以不會被誤判成葉子而重複計算
                if(previous->left == nullptr && previous->right == nullptr) {
                    totalSum += previous->val;
                }

                // 找 currentNode 的中序前驅：左子樹中最右邊的節點
                // 停止條件：right 是空的（第一次），或 right 已經指回 currentNode（第二次）
                while(previous->right != nullptr && previous->right != currentNode) {
                    previous = previous->right;
                }

                if(previous->right == nullptr) {
                    // 第一次到達：先接線索（前驅的 right 指回 currentNode），
                    // 這樣走完左子樹後才有路可以回來
                    previous->right = currentNode;
                    currentNode = currentNode->left;
                } else {
                    // 第二次到達：左子樹已經走完，拆掉線索還原樹的結構，改走右子樹
                    previous->right = nullptr;
                    currentNode = currentNode->right;
                }
            }
        }
        return totalSum;
    }
};
```

## 解法：tree-bfs-level-order

**難度:** 🟢 E　**標籤:** Tree Bfs Level Order　**時間:** O(N)　**空間:** O(W)　**熟悉度:** 🟠 再練習　**官方解答:** ✅ 已同步官方解答

```cpp
/**
 * 題目：404. Sum of Left Leaves
 * 難度：Easy
 * 時間複雜度：O(N)
 * 空間複雜度：O(W)，W 為樹的最大寬度（最差 O(N)）
 * 分類主題：tree-bfs-level-order
 *
 * 解法說明：
 * 用 BFS 逐層走訪整棵樹，每個節點在入佇列時一併記錄「自己是不是父節點的左子」。
 * 出佇列時，若該節點是左子且沒有任何子節點（葉子），就把它的值加進總和。
 */
class Solution {
public:
    int sumOfLeftLeaves(TreeNode* root) {
        if(!root) return 0;

        // 佇列存 (節點, 是否為左子)，因為節點本身不知道自己是左或右，
        // 必須在入佇列時由父節點的角度標記
        queue<pair<TreeNode*, bool>> q;
        q.push({root, false});  // 根節點沒有父節點，不算左子
        int sum = 0;

        while(!q.empty()) {
            // 一次處理完當前這一層（先固定這層的節點數，避免新加入的下一層被混進來）
            // 本題其實不需要分層，這個 for 只是標準的 level-order 寫法
            for(int i = q.size() - 1; i >= 0; i--) {
                auto [node, isLeft] = q.front();
                q.pop();

                // 左葉子的條件：是左子 且 沒有左右子節點
                if(isLeft && !node->left && !node->right)  sum += node->val;

                // 子節點入佇列，順便標記身分：左子 true、右子 false
                if(node->left) q.push({node->left, true});
                if(node->right) q.push({node->right, false});
            }
        }
        return sum;
    }
};
```

>**不能放在for loop 外面，不然無法處理到中間的左葉節點**<br>

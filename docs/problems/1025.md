```cpp
/**
 * 題目：1025. Divisor Game (除數博弈)
 * 難度：簡單 (Easy)
 * 描述：Alice 和 Bob 輪流操作黑板上的整數 n（Alice 先手）：選擇任一滿足
 *       0 < x < n 且 n % x == 0 的整數 x，並把 n 替換成 n - x。無法移動的玩家輸掉遊戲。
 *       假設雙方都採取最佳策略，判斷 Alice 是否能獲勝。
 *
 * 時間複雜度：O(1)
 * 空間複雜度：O(1)
 *
 * 解法思路：
 * 1. 關鍵觀察：n 為偶數時 Alice 必勝 (Even n is a Winning Position)：
 *    - 若 n 是偶數，Alice 永遠可以選擇 x = 1（1 一定整除任何 n），
 *      把 n 變成奇數，丟給 Bob 面對奇數局面。
 * 2. n 為奇數時的性質：奇數的因數必為奇數 (Odd n's Divisors are Odd)：
 *    - 若 n 是奇數，n 的所有因數 x 也都必定是奇數，
 *      因此 n - x（奇數減奇數）結果必定是偶數，代表奇數局面的玩家，
 *      不管怎麼選，都只能把偶數局面留給對手。
 * 3. 歸納出「偶勝奇敗」的規律：
 *    - 這形成一個穩定的循環：偶數局面的玩家，永遠能把奇數局面丟給對手；
 *      奇數局面的玩家，則永遠被迫把偶數局面留給對手。
 *    - 由於遊戲最終會在 n=1（無法移動，輪到的玩家輸）結束，
 *      透過歸納法可以證明：n 為偶數時，先手的玩家（在本題中即為 Alice）必勝。
 * 4. 結論化簡成一行判斷式：
 *    - 直接回傳 n % 2 == 0，即為「n 是偶數時 Alice 必勝」的結論。
 */
class Solution {
public:
    bool divisorGame(int n) {
        return (n % 2 == 0);
    }
};
```

> **偶勝奇敗，n是偶數時Alice可以選擇1，必勝**<br>
> **n 為奇數時的性質：奇數的因數必為奇數**<br>

```cpp
/*
 * 題目：1025. Divisor Game
 * 難度：Easy
 * 描述：Alice 與 Bob 輪流玩遊戲，Alice 先手。黑板上有整數 n，輪到的人選一個 x，滿足 0 < x < n 且 n % x == 0，然後把 n 換成 n - x。無法行動的人輸。兩人都採最佳策略，回傳 Alice 是否會贏。
 * 分類主題：dp-game-theory
 * 技巧：必勝必敗狀態 DP, 取補集 (對手必敗則我必勝), 找到必勝走法即 break
 * 時間複雜度：O(n^2)
 * 空間複雜度：O(n)
 * 解法思路：
 *   1. dp[i] 表示輪到自己、黑板上是 i 時，自己能不能贏
 *   2. dp[1] = false：沒有合法的 x，直接輸
 *   3. 轉移：只要存在一個 x 使 i % x == 0 且 dp[i - x] 為 false（對手拿到 i - x 會輸），dp[i] 就為 true
 *   4. 由小到大填表，答案是 dp[n]
 *   5. 數學規律：偶數必勝（選 1 讓對手拿奇數），奇數必敗（奇數的因數都是奇數，只能讓對手拿到偶數），所以也可以直接回傳 n % 2 == 0，時間降為 O(1)
 */
class Solution {
public:
    bool divisorGame(int n) {
        // dp[i]：輪到自己、數字為 i 時，自己能不能贏（預設都先當作會輸，dp[1] 本來就是 false）
        vector<bool> dp(n + 1, false);
        for (int i = 2; i <= n; i++)
            for (int x = 1; x < i; x++)
                // x 是 i 的因數，且拿走 x 之後對手面對 i - x 會輸 → 自己這一步必勝
                if (i % x == 0 && !dp[i - x]) { dp[i] = true; break; }
        return dp[n];
    }
};
```

```cpp
/**
 * 題目：342. Power of Four
 * 難度：簡單 (Easy)
 * 描述：判斷一個整數是否為 4 的冪次方。
 *
 * 時間複雜度：O(logN)
 * 空間複雜度：O(1)
 *
 * 解法思路：
 * （迴圈除以 4）：
 * 反覆除以 4，最後若等於 1 則是 4 的冪次方。
 */
class Solution {
public:
    bool isPowerOfFour(int n) {
        if(n <= 0) return false;
        while(n % 4 == 0) n /= 4;
        return n == 1;
    }
};
```

```cpp
/**
 * 題目：342. Power of Four
 * 難度：簡單 (Easy)
 * 描述：判斷一個整數是否為 4 的冪次方。
 *
 * 時間複雜度：O(1)
 * 空間複雜度：O(1)
 *
 * 解法思路：
 * （位元運算）：
 * 4 的冪次方必須是 2 的冪次方（只有一個 1 bit），且 n-1 必須能被 3 整除。
 */
class Solution {
public:
    bool isPowerOfFour(int n) {
        if(n <= 0) return false;
        return (n & (n - 1)) == 0 && n % 3 == 1;
    }
};
```

```cpp
// 題目：342. Power of Four
// 難度：Easy
// 時間複雜度：O(1)
// 空間複雜度：O(1)
// 分類主題：math
// 技巧：對數判斷（log₂(n) 是偶數 ⇔ n 是 4 的冪）

class Solution {
public:
    bool isPowerOfFour(int n) {
        // 4 的冪一定是正數；0 與負數的 log 無意義（NaN / -inf），先排除
        if(n <= 0) return false;

        // 換底公式：log₂(n) = ln(n) / ln(2)
        // 若 n = 4^k = 2^(2k)，則 log₂(n) = 2k，一定是偶數
        // fmod(x, 2) 是浮點數取餘數（C++ 的 % 不能用在 double），
        // 餘 0 表示 log₂(n) 是偶數
        // 若 n 不是 2 的冪（如 3、5），log₂(n) 不是整數，餘數不會剛好為 0；
        // 若 n 是 2 的奇數次方（如 2、8、32），log₂(n) 是奇數，餘數為 1
        return (fmod(log(n) / log(2), 2) == 0);
    }
};
```

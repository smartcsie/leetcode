```cpp
/**
 * 題目：171. Excel Sheet Column Number
 * 難度：簡單 (Easy)
 * 描述：將 Excel 的欄位名稱 (如 "A", "AB") 轉換為對應的數字。
 *
 * 時間複雜度：O(N)
 * 空間複雜度：O(1)
 *
 * 解法思路：
 * 1. 將 Excel 欄位視為 26 進位系統。
 * 2. 遍歷字串，每一位數的值為 `sum * 26 + (當前字母轉數字)`。
 * 3. 使用 `std::accumulate` 進行累加，將字串從左到右摺疊計算出最終結果。
 */

#include <string>
#include <numeric>

class Solution {
public:
    int titleToNumber(std::string columnTitle) {
        // 使用 accumulate 進行左摺疊運算 (Left Fold)
        return std::accumulate(columnTitle.begin(), columnTitle.end(), 0, [](int sum, char c) {
            return sum * 26 + (c - 'A' + 1);
        });
    }
};
```

## 解法：math-digit-decomposition

**難度:** 🟢 E　**標籤:** Math Digit Decomposition　**時間:** O(n)　**空間:** O(1)　**熟悉度:** 🟢 熟練　🏆 **最佳解**　**官方解答:** ✅ 已同步官方解答

```cpp
// 題目：171. Excel Sheet Column Number
// 難度：Easy
// 時間複雜度：O(n)，n = columnTitle 的長度
// 空間複雜度：O(1)
// 分類主題：math / string
// 技巧：26 進位轉換（A~Z 對應 1~26，沒有 0），類似「字串轉整數」的 sum = sum * base + digit

class Solution {
public:
    int titleToNumber(string columnTitle) {
        int sum = 0;
        for(const char& c : columnTitle) {
            // 由左到右處理每個字母，等同 26 進位：
            //   sum * 26：把目前累積的值整體進一位（像十進位的 sum * 10）
            //   c - 'A' + 1：這個字母代表的數字，'A' = 1、'B' = 2、... 'Z' = 26
            // 注意是 1~26，不是 0~25，所以要 + 1
            // 例："AB" -> 先 'A'：0 * 26 + 1 = 1；再 'B'：1 * 26 + 2 = 28
            //     "ZY" -> 'Z'：26；'Y'：26 * 26 + 25 = 701
            sum = sum * 26 + (c - 'A' + 1);
        }
        // 題目保證結果落在 int 範圍內（最大為 "FXSHRXW" = 2^31 - 1），不會溢位
        return sum;
    }
};
```

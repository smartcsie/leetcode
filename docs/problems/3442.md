```cpp
/**
 * 題目：3442. Maximum Difference Between Even and Odd Frequency I
 * 難度：簡單 (Easy)
 * 描述：計算字串中「奇數頻率的最大值」與「偶數頻率的最小值」之差。
 *
 * 時間複雜度：O(N)
 * 空間複雜度：O(1)
 *
 * 解法思路：
 * 1. 使用長度為 26 的陣列記錄每個字母的出現頻率。
 * 2. 遍歷字串統計頻率。
 * 3. 再次遍歷陣列，分別維護 `maxOdd` 與 `minEven` 的記錄。
 * 4. 最後返回 `maxOdd - minEven`。
 */

class Solution {
public:
    int maxDifference(string s) {
        array<uint8_t, 26> counts{};
        int odd = 0 , even = 101;
        for(const char& c : s) {
            counts[c - 'a']++;
        }
        for(int i = 0; i < 26; i++) {
            if(counts[i] == 0) continue;
            if(counts[i] & 1) odd = max(odd, (int)counts[i]);
            else even = min(even, (int)counts[i]);
        }
        return odd - even;
    }
};
```

> **用 max even frequency 減去 min even frequency**<br>

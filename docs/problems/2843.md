/**
 * 題目：2843. Count Symmetric Integers
 * 難度：簡單 (Easy)
 * 描述：計算 [low, high] 區間內有多少「對稱整數」。
 * 定義：位數為偶數，且將數字對半分成兩部分，兩部分數位和相等。
 *
 * 時間複雜度：O(N*D)
 * 空間複雜度：O(1)
 *
 * 解法思路：
 * 1. 遍歷區間內的每個整數。
 * 2. 判斷數字位數：
 * - 2 位數：判斷 (n/10) == (n%10)
 * - 4 位數：前兩位數和 == 後兩位數和
 * 3. 注意：位數為奇數的整數 (例如 100-999) 永遠不對稱，無需考慮。
 */

class Solution {
public:
    int countSymmetricIntegers(int low, int high) {
        int count = 0;
        for(int x = low; x <= high; x++) {
            if(!(x >= 11 && x <=99)&& !(x >= 1000 && x <=9999)) continue;
            else if(x >= 11 && x <= 99 && (x/10 == x % 10)) count++;
            else if(x >= 1000 && x <=9999) {
                int x1 = x / 100;
                int x2 = x % 100;
                if((x1/10 + x1 % 10) == (x2/10 + x2 % 10)) count++;
            }
        }
        return count;
    }
};

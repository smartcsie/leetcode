```cpp
/**
 * 題目：3736. Minimum Moves to Equal Array Elements III
 * 難度：Easy
 * 時間複雜度：O(N)
 * 空間複雜度：O(1)
 * 分類主題：array-min-max
 *
 * 解法說明：
 * 每次操作只能讓「一個」元素 +1，所以數字只能往上加，最後所有元素必須等於某個目標值 T，
 * 而 T 最小只能是原陣列的最大值 max（比 max 小的目標值，max 那個元素降不下來）。
 * 目標值取 max 時，每個元素 x 需要補 (max - x) 次，總次數：
 *   Σ(max - x) = n * max - sum
 * 所以只要掃描一次，同時求出總和與最大值即可。
 */
class Solution {
public:
    int minMoves(vector<int>& nums) {
        int sum = 0;   // 所有元素總和
        int mx = 0;    // 最大值（題目的元素為正整數，初始值 0 即可）
        int n = nums.size();

        // 一趟掃描同時累加總和與更新最大值
        for(const int& x : nums) {
            sum += x;
            mx = max(mx, x);
        }

        // 全部補到 mx 所需的總次數 = n * mx - sum
        return n * mx - sum;
    }
};
```

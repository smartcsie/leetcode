```cpp
/**
 * 題目：160. Intersection of Two Linked Lists (相交鏈結串列)
 * 難度：簡單 (Easy)
 * 描述：找到兩個單鏈結串列相交的起始節點。若不相交，回傳 nullptr。
 *
 * 時間複雜度：O(N+M)
 * 空間複雜度：O(1)
 *
 * 解法思路：
 * (雙指標)：
 * 1. 設置兩個指標 listA 和 listB 分別從 headA 和 headB 開始走。
 * 2. 邏輯核心：當 listA 走到盡頭 (nullptr) 時，將其指向 headB；
 * 同理，listB 走到盡頭時，指向 headA。
 * 3. 為什麼會相遇？
 * - 假設 A 的獨特長度為 a，B 的獨特長度為 b，交點後的公共長度為 c。
 * - listA 走過的路徑為：a + c + b
 * - listB 走過的路徑為：b + c + a
 * - 因為 a + c + b = b + c + a，它們一定會在交點處相遇。
 * 4. 如果兩者不相交，它們會在最後同時變成 nullptr 而跳出迴圈 (listA == listB == nullptr)。
 */

class Solution {
public:
    ListNode *getIntersectionNode(ListNode *headA, ListNode *headB) {
        ListNode* listA = headA;
        ListNode* listB = headB;
        while(listA != listB) {
            listA = listA ? listA->next : headB;
            listB = listB ? listB->next : headA;
        }
        return listA;
    }
};
```

```cpp
/**
 * 題目：160. Intersection of Two Linked Lists
 * 難度：Easy
 * 時間複雜度：O(M + N)
 * 空間複雜度：O(M + N)
 * 分類主題：linked-list
 *
 * 解法說明：
 * 兩條鏈表若相交，相交點之後的節點會完全共用（是同一批節點，不只是值相同）。
 * 因此把兩條鏈表分別壓進 stack，從尾端開始往回比對，
 * 尾端連續相同的節點就是共用部分，最後一個相同的節點即為相交起點。
 */
class Solution {
public:
    ListNode *getIntersectionNode(ListNode *headA, ListNode *headB) {
        stack<ListNode*> st1, st2;

        // 把 A 鏈表所有節點由頭到尾壓入 stack（top 會是 A 的尾端）
        ListNode * cur = headA;
        while(cur) {
            st1.push(cur);
            cur = cur->next;
        }

        // 同樣處理 B 鏈表
        cur = headB;
        while(cur) {
            st2.push(cur);
            cur = cur->next;
        }
        
        // 從尾端往回比對：比較的是「節點指標」而非節點值
        // 只要兩邊 top 是同一個節點，就代表還在共用段內，持續記錄並彈出
        // 一旦不同（或任一 stack 空了）就代表已退到共用段的起點之前
        ListNode *intersection = nullptr;
        while(!st1.empty() && !st2.empty() && st1.top() == st2.top()) {
            intersection = st1.top();  // 目前已知的最前面一個共用節點
            st1.pop();
            st2.pop();
        }

        // 若尾端就不同（沒有共用段），intersection 維持 nullptr，代表不相交
        return intersection;
    }
};
```

## 解法：hash-table-existence

**難度:** 🟢 Easy　**標籤:** Hash Table Existence　**時間:** O(M + N)　**空間:** O(M)　**熟悉度:** 🟢 熟練

```cpp
/**
 * 題目：160. Intersection of Two Linked Lists
 * 難度：Easy
 * 時間複雜度：O(M + N)
 * 空間複雜度：O(M)
 * 分類主題：hash-table-existence
 *
 * 解法說明：
 * 若兩條鏈表相交，相交點之後的節點是「同一批節點」（指標位址相同）。
 * 先把 A 鏈表的所有節點指標存進 hash set，
 * 再走訪 B 鏈表，第一個出現在 set 裡的節點就是相交起點。
 */
class Solution {
public:
    ListNode *getIntersectionNode(ListNode *headA, ListNode *headB) {
        // 存的是「節點指標」而非節點值，
        // 因為值相同的節點不代表是同一個節點，只有位址相同才算相交
        unordered_set<ListNode*> seen;

        // 走訪 A 鏈表，把每個節點記錄下來
        ListNode * cur = headA;
        while(cur) {
            seen.insert(cur);
            cur = cur->next;
        }

        // 走訪 B 鏈表，由頭開始檢查
        // 第一個在 set 裡找到的節點，就是 B 這邊最早進入共用段的節點，即相交起點
        cur = headB;
        while(cur) {
            if(seen.contains(cur)) return cur;  // contains 為 C++20 語法
            cur = cur->next;
        }

        // B 鏈表整條都沒有節點出現在 A 裡 → 不相交
        return nullptr;
    }
};
```

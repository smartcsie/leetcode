# 1609. Even Odd Tree

[在 LeetCode 上查看](https://leetcode.com/problems/even-odd-tree/)

**難度:** 🟡 Medium　**標籤:** Tree, BFS　**時間:** O(N)　**空間:** O(W)

```cpp
/**
 * 註解：待補充
 */
class Solution {
public:
    
    bool isEvenOddTree(TreeNode* root) {
        queue<TreeNode*> q({root});
        int level = 0;
        while(!q.empty()) {
            int prev = -1;
            for(int i = q.size(); i > 0 ; i --) {
                TreeNode* node = q.front();
                q.pop();
                if(level%2 == 0) {
                    if(node->val %2 ==0) {
                        return false;
                    } 
                } else {
                    if(node->val %2 ==1) {
                        return false;
                    } 
                }
                if(prev != -1) {
                    if(level%2 == 0 && prev >= node->val) return false;
                    if(level%2 == 1 && prev <= node->val) return false;
                }
                prev = node->val;
                if(node->left) q.push(node->left);
                if(node->right) q.push(node->right);
            }
            level++;
        }
        return true;
    }
};
```

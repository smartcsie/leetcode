```cpp
/*
 * 題目：1941. Check if All Characters Have Equal Number of Occurrences
 * 難度：Easy
 * 描述：給定字串 s，若 s 中每個出現過的字元，出現次數都相同，回傳 true，否則回傳 false。
 * 分類主題：hash-table-counting
 * 技巧：固定大小計數陣列, 以第一個字元的次數當基準
 * 時間複雜度：O(n + 26)，n 為字串長度
 * 空間複雜度：O(1)（固定 26 格計數陣列）
 * 解法思路：
 *   1. 用長度 26 的陣列統計每個小寫字母的出現次數
 *   2. s 一定有出現過的字元，取 s[0] 的次數當作基準 freq
 *   3. 掃過 26 個計數：沒出現過的（0）略過，出現過的必須等於 freq，否則回傳 false
 *   4. 全部通過回傳 true
 */
class Solution {
public:
    bool areOccurrencesEqual(string s) {
        // 26 個小寫字母的出現次數，{} 初始化為全 0
        array<uint16_t, 26> counts{};
        for(const char& c : s) counts[c - 'a']++;

        // 以第一個字元的次數當基準（s 非空，所以它一定出現過）
        int freq = counts[s[0] - 'a'];

        for(const int& count : counts) {
            // 沒出現的字母（0 次）不用比；出現過的次數必須與基準相同
            if(count != 0 && count != freq) return false;
        }
        return true;
    }
};
```

/**
 * 題目：421. Maximum XOR of Two Numbers in an Array (陣列中兩個數字的最大異或值)
 * 難度：中等 (Medium)
 * 描述：給定一個整數陣列 nums，找出 nums[i] XOR nums[j] 的最大結果，其中 0 <= i <= j < n。
 *
 * 時間複雜度：O(N*L)
 * 空間複雜度：O(N)
 *
 * 解法思路：
 * 1. 逐位元建構前綴遮罩 (Prefix Masking)：
 *    - 從最高位元往低位元掃描，每次將該位元加入 prefixMask，取得目前所有數字的「前綴」（高位共同部分）。
 * 2. 貪婪嘗試候選答案 (Greedy Candidate)：
 *    - 假設答案在這一位元可以是 1（candidate = ans | 1 << i），檢查是否存在兩個前綴的 XOR 恰好等於 candidate。
 * 3. 集合查找驗證 (Hash Set Lookup)：
 *    - 若 prefix ^ candidate 存在於 prefixes 集合中，代表這一位元可以達成，將 ans 更新為 candidate。
 */

class Solution {
public:
    int findMaximumXOR(vector<int>& nums) {
        int mx = *max_element(nums.begin(), nums.end());
        int bits = bit_width(static_cast<unsigned int>(mx));
        int bitMask = 0;
        int ans = 0;
        for(int i = bits - 1; i >= 0; i--) {
            bitMask |= 1 << i;
            unordered_set<int> prefixs;
            for(const int& num : nums) {
                prefixs.insert(num & bitMask);
            }
            int candidate = ans | 1 << i;
            for(const int& prefix : prefixs) {
                //這裡原本是 存在兩數num1, num2 之prefix 有辦法
                // prefix1 ^ prefix2 == candidate(最佳) 則更新 ans prefix
                // 等價 prefixs.contains(prefix ^ candidate)
                if(prefixs.contains(prefix ^ candidate)) {
                    ans = candidate;
                    break;
                }
            }
        }
        return ans;
    }
};

    /**
     * 題目：421. Maximum XOR of Two Numbers in an Array
     * 難度：Medium
     * 描述：給定非負整數陣列 nums，回傳 nums[i] XOR nums[j] 的最大值（0 <= i <= j < n）。
     *
     * 分類主題：trie
     * 技巧：二進位 Trie, 由高位到低位貪心選相反位元, 邊插入邊查詢, 以最大值的位元長度決定層數
     *
     * 時間複雜度：O(N · L)，L 為最大值的二進位位數，最多 31
     * 空間複雜度：O(N · L)，Trie 最多 N · L 個節點
     *
     * 解法思路：二進位 Trie + 貪心
     * 1. XOR 的結果要最大，就要讓高位元盡量是 1。
     *    對固定的 num，從最高位開始，每一位都優先走「和 num 該位元相反」的路，
     *    走得通就讓結果的該位元為 1，走不通才退而求其次走相同位元。
     * 2. 把每個數字的二進位（固定 L 位，由高到低）當作一條路徑存進 Trie，
     *    每個節點只有兩個子節點，代表 0 與 1。
     *    L 取最大值的二進位位數，所有數字補齊到同樣長度，路徑才能對齊。
     * 3. 對每個 num 同時做兩件事（用兩個指標走）：
     *    - node：把 num 插入 Trie。
     *    - xorNode：在 Trie 中貪心尋找能和 num 配出最大 XOR 的路徑。
     * 4. 邊插入邊查詢不會漏掉任何配對：
     *    兩個數 a、b 中較晚處理的那個，查詢時另一個已經在 Trie 裡了。
     *    num 也可能和自己配對，XOR 為 0，不影響答案（答案至少是 0）。
     * 5. 查詢時 xorNode 不會走到空節點：
     *    xorNode 一定停在某條已存在的完整路徑上（包含剛插入的 num 自己），
     *    若相反位元的子節點不存在，則相同位元的子節點必然存在。
     * 6. 特例：最大值為 0 代表全部是 0，答案是 0，直接回傳。
     */
    struct TrieNode {
        TrieNode* children[2] = {nullptr, nullptr};  // children[0]、children[1] 對應位元 0、1
    };
    
    class Solution {
    public:
        int findMaximumXOR(std::vector<int>& nums) {
            int maxNum = *std::max_element(nums.begin(), nums.end());
            if (maxNum == 0) return 0;  // 全部是 0，任何 XOR 都是 0
    
            // 計算二進位長度 L (等價 Java 的 Integer.toBinaryString(maxNum).length())
            // std::bit_width 需要 C++20 與 <bit>
            int L = std::bit_width(static_cast<unsigned int>(maxNum));
            int maxXor = 0;
    
            TrieNode* root = new TrieNode();
    
            for (int num : nums) {
                TrieNode* node = root;     // 插入用的指標
                TrieNode* xorNode = root;  // 查詢用的指標
                int currXor = 0;           // num 與 Trie 中某數配出的最大 XOR
    
                for (int i = L - 1; i >= 0; i--) {
                    int bit = (num >> i) & 1;  // num 的第 i 位
                    int toggledBit = bit ^ 1;  // 相反位元，走這條路 XOR 該位為 1
    
                    // 1. 建立 / 插入當前數字到 Trie
                    if (node->children[bit] == nullptr) {
                        node->children[bit] = new TrieNode();
                    }
                    node = node->children[bit];
    
                    // 2. 貪婪尋找能夠產生最大 XOR (相反位元) 的路徑
                    if (xorNode->children[toggledBit] != nullptr) {
                        currXor |= (1 << i);                    // 這一位能湊出 1
                        xorNode = xorNode->children[toggledBit];
                    } else {
                        xorNode = xorNode->children[bit];       // 只能走相同位元，這一位是 0
                    }
                }
    
                maxXor = std::max(maxXor, currXor);
            }
    
            return maxXor;
        }
    };

```cpp
/**
 * 題目：2784. Check if Array is Good (檢查陣列是否有效)
 * 難度：簡單 (Easy)
 * 描述：給定一個整數陣列 nums。如果該陣列是 [1, 2, ..., n-1, n, n] 的一個排列，則稱其為「好陣列」。
 * 也就是說，如果最大值為 n，則陣列長度必須為 n + 1，且 1 到 n-1 各出現一次，n 出現兩次。
 *
 * 時間複雜度：O(N)
 * 空間複雜度：O(N)
 */

class Solution {
public:
    bool isGood(vector<int>& nums) {
        array<uint8_t, 201> counts{};
        int n = nums.size();
        for(const int& x : nums) {
            if(x >= n ) return false;
            if(x == n - 1 && counts[x] >= 2) return false;
            if(x < n - 1 && counts[x] >= 1) return false;
            counts[x]++;
        }
        return true;
    }
};
```

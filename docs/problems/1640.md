```cpp
/**
 * 題目：1640. Check Array Formation Through Concatenation
 * 難度：簡單 (Easy)
 * 描述：給定一個整數陣列 arr 和二維陣列 pieces，判斷能否從 pieces 中選出
 * 若干個子陣列（每個最多用一次），按順序拼接成 arr。
 *
 * 時間複雜度：O(N)
 * 空間複雜度：O(N)
 *
 * 解法思路：
 * （Hash Map 以首元素快速定位 piece）：
 * 1. 建立 headToIdx 陣列，以每個 piece 的首元素為 key，記錄其在 pieces 的索引。
 *    （題目保證所有數字唯一，不會有衝突）
 * 2. 遍歷 arr，對每個位置查找以 arr[i] 為首的 piece：
 *    - 找不到：回傳 false。
 *    - 找到後逐一比對 piece 的每個元素與 arr 是否一致。
 * 3. 全部比對成功則回傳 true。
 *
 * 注意：
 * 使用 array<int, 101> 取代 unordered_map，
 * 因為數字範圍在 [1, 100]，固定大小陣列存取更快。
 */
class Solution {
public:
    bool canFormArray(vector<int>& arr, vector<vector<int>>& pieces) {
        int n = arr.size();
        array<int, 101> headToIdx;
        headToIdx.fill(-1);

        // 建立首元素 → piece 索引的映射
        for (int i = 0; i < (int)pieces.size(); i++)
            headToIdx[pieces[i].front()] = i;

        int i = 0;
        while (i < n) {
            int pIdx = headToIdx[arr[i]];
            if (pIdx == -1) return false;  // 找不到對應的 piece
            for (const int& x : pieces[pIdx])
                if (x != arr[i++]) return false;  // piece 內容與 arr 不符
        }
        return true;
    }
};
```

> **用一組  array<int, 101> 紀錄每組vector第一個int和該組vector idx的對應 **<br>
> **對arr使用index i，如果arr[i]在headToIdx[arr[i]]中找到，取出來，每個int開始比對，i往後走**<br>

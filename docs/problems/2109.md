```cpp
/**
 * 題目：2109. Adding Spaces to a String (向字串添加空格)
 * 難度：中等 (Medium)
 * 描述：在給定的空間索引處插入空格，並回傳新的字串。
 *
 * 時間複雜度：O(N+M)
 * 空間複雜度：O(N+M)
 *
 * 解法思路：
 * 1. 使用 `res` 字串存放結果。為了效能，使用 `reserve()` 預先分配足夠空間。
 * 2. 使用指標 `j` 追蹤 `spaces` 陣列的索引，`i` 追蹤原字串 `s` 的索引。
 * 3. 遍歷過程中，若 `i` 達到 `spaces[j]` 的位置，則先加入空格並移動 `j`。
 * 4. 最後加入當前字元 `s[i]`。
 */

class Solution {
public:
    string addSpaces(string s, vector<int>& spaces) {
        string ans;
        for(int cIdx = 0, sIdx = 0; cIdx < s.size(); cIdx++) {
            if(sIdx < spaces.size() && cIdx == spaces[sIdx]) {
                ans.push_back(' ');
                sIdx++;
            }
            ans.push_back(s[cIdx]);
        }
        return ans;
    }
};
```

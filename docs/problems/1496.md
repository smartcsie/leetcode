```cpp
/**
 * 題目：1496. Path Crossing
 * 難度：簡單 (Easy)
 * 描述：給定一個由 N/S/E/W 組成的路徑字串，判斷路徑是否會經過已走過的點。
 *
 * 時間複雜度：O(N*logN)
 * 空間複雜度：O(N)
 *
 * 解法思路：
 * （set 記錄走過的座標）：
 * 1. 用 set 儲存所有走過的座標，初始放入原點 {0, 0}。
 * 2. 每走一步更新當前座標，檢查是否已在 set 中。
 * 3. 若已存在代表路徑交叉，回傳 true；否則將新座標加入 set。
 *
 * 注意：
 * 使用 set<vector<int>> 而非 unordered_set，
 * 因為 vector<int> 沒有預設 hash function，
 * 若要用 unordered_set 需自訂 hash。
 */

class Solution {
public:
    bool isPathCrossing(string path) {
        unordered_set<int> travsal;
        int x = 0 , y = 0;
        travsal.insert(0);
        for(const char c : path) {
            if(c == 'N') y++;
            else if(c == 'S') y--;
            else if(c == 'E') x++;
            else if(c == 'W') x--;
            int key = x * 20001 + y;
            if(travsal.contains(key)) return true;
            travsal.insert(key);
        }
        return false;
    }
};
```

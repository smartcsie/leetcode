```cpp
/**
 * 題目：78. Subsets
 * 難度：Medium
 * 描述：給定元素互不相同的整數陣列 nums，回傳所有可能的子集（冪集），
 *       解集不能包含重複的子集，順序不拘。
 *
 * 分類主題：backtracking-combination
 * 技巧：迭代式逐步擴展子集, 每處理一個數字子集數量翻倍, reserve 預先配置空間
 *
 * 時間複雜度：O(n * 2^n)
 * 空間複雜度：O(1)（不計輸出，沒有遞迴也沒有暫存陣列）
 *
 * 解法思路：迭代擴展
 * 1. 一開始 ans 只有空集合。
 * 2. 每處理一個數字 x，就把「目前所有子集」各複製一份並在尾端加上 x，
 *    新子集接在 ans 後面，子集數量剛好翻倍。
 * 3. 先固定 size = ans.size() 再迴圈，本輪新增的子集不會在同一輪被再次擴展。
 * 4. 處理完 n 個數字，共 2^n 個子集。
 * 5. 先 reserve(1 << n)，避免 push_back 過程中反覆搬移整個答案。
 */
class Solution {
public:
    vector<vector<int>> subsets(vector<int>& nums) {
        vector<vector<int>> ans = {{}};  // 一開始只有空集合
        ans.reserve(1 << nums.size());

        for (int x : nums) {
            int size = ans.size();  // 本輪開始前的子集數量
            for (int j = 0; j < size; j++) {
                ans.push_back(ans[j]);  // 複製現有子集
                ans.back().push_back(x);  // 再加上 x，形成新子集
            }
        }
        return ans;
    }
};
```

```cpp
/**
 * 題目：78. Subsets
 * 難度：中等 (Medium)
 * 描述：給你一個含不同元素的整數陣列 nums，請你返回該陣列所有可能的子集（冪集）。
 * 解集不能包含重複的子集。你可以按任意順序返回解集。
 *
 * 時間複雜度：O(N*2^N)
 * 空間複雜度：O(N)
 *
 * 解法思路：
 * （位元遮罩窮舉法 Bitmask Exhaustion）：
 * 1. 冪集的總數：
 * - 一個長度為 `n` 的集合，其子集總數為 $2^n$ 個。我們可以用一個整數 `mask` 從 `0` 窮舉到 `2^n - 1`。
 * 2. 利用位元對應元素選擇：
 * - 對於每一個 `mask`，其二進位的第 `i` 位若為 `1`，代表將 `nums[i]` 選入當前的子集 `cur` 中；若為 `0` 則不選。
 * 3. 收集結果：
 * - 將每一個建構好的子集 `cur` 放入結果矩陣 `res` 中，直到所有 $2^n$ 種組合遍歷完畢。
 */

class Solution {
public:
    std::vector<std::vector<int>> subsets(std::vector<int>& nums) {
        int n = nums.size();
        std::vector<std::vector<int>> res;
        // 總共有 2^n 種選擇組合 (從 0 到 2^n - 1)
        for (int mask = 0; mask < (1 << n); mask++) {
            std::vector<int> cur;
            // 檢查 mask 的每一個位元，決定是否將 nums[i] 納入子集
            for (int i = 0; i < n; i++) {
                if (mask & (1 << i)) {
                    cur.push_back(nums[i]);
                }
            }
            res.push_back(cur);
        }
        return res;
    }
};
```

```cpp
/**
 * 題目：78. Subsets
 * 難度：Medium
 * 描述：給定元素互不相同的整數陣列 nums，回傳所有可能的子集（冪集），
 *       解集不能包含重複的子集，順序不拘。
 *
 * 分類主題：bit-bitmask-subset-enumeration
 * 技巧：位元遮罩列舉所有子集, mask 的第 i 位元代表是否選 nums[i], 直接用整數位元運算
 *
 * 時間複雜度：O(n * 2^n)
 * 空間複雜度：O(n)，暫存的一個子集（不計輸出）
 *
 * 解法思路：位元遮罩
 * 1. n 個元素共有 2^n 種選法，用整數 mask 的第 i 位元表示「是否選 nums[i]」。
 * 2. mask 從 0 跑到 2^n - 1，剛好列舉所有選法，每個 mask 對應一個子集。
 * 3. 對每個 mask 掃過 n 個位元，位元為 1 就把 nums[i] 放進子集。
 * 4. 直接用 mask >> i & 1 判斷位元，比先轉成字串再逐字元比對快很多，
 *    1 << n 也比 pow(2, n) 好（pow 是浮點運算）。
 */
class Solution {
public:
    vector<vector<int>> subsets(vector<int>& nums) {
        int n = nums.size();
        vector<vector<int>> ans;
        ans.reserve(1 << n);

        for (int mask = 0; mask < (1 << n); mask++) {
            vector<int> v;
            for (int i = 0; i < n; i++) {
                if (mask >> i & 1) v.push_back(nums[i]);  // 第 i 位元為 1：選 nums[i]
            }
            ans.push_back(v);
        }
        return ans;
    }
};
```

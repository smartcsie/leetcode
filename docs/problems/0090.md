/**
 * 題目：90. Subsets II
 * 難度：Medium
 * 描述：給定可能含重複數字的整數陣列 nums，回傳所有可能的子集（冪集），
 *       解集不能包含重複的子集，順序不拘。
 *
 * 分類主題：backtracking-combination
 * 技巧：先排序讓相同數字相鄰, 迭代式逐步擴展子集, 遇到重複數字只擴展上一輪新增的子集
 *
 * 時間複雜度：O(n * S)，S 為不重複子集的數量，最壞（全部相異）S = 2^n
 *   每個子集建立時要複製一次，花 O(n)
 * 空間複雜度：O(n)，暫存的一個子集（不計輸出）
 *
 * 解法思路：迭代擴展 + 重複數字只接在上一輪新增的子集後面
 * 1. 先排序：相同數字排在一起，才能用「與前一個比較」判斷是否重複。
 * 2. 一開始 ans 只有空集合。每處理一個數字 nums[i]，
 *    就把它加到若干個現有子集的尾端，產生新子集。
 * 3. nums[i] 與前一個不同：替「所有」現有子集加上 nums[i]，
 *    這是標準的子集生成方式，每輪子集數量翻倍。
 * 4. nums[i] 與前一個相同：只替「上一輪新增的那批」加上 nums[i]。
 *    原因：更早的子集加上這個數字，在上一輪（處理前一個相同數字時）已經產生過了，
 *    再加一次會得到重複的子集。
 *    例如 [1,2,2]：第二個 2 若也擴展 [] 與 [1]，會得到 [2]、[1,2]，和第一個 2 的結果重複。
 * 5. lastStart 記錄「本輪新增的子集在 ans 中的起點」，下一輪遇到重複數字時，
 *    就從這個索引開始擴展，一直到本輪開始前的 ans 大小為止。
 * 6. 先固定 to = ans.size() 再迴圈，確保本輪新增的子集不會在同一輪被再次擴展。
 */
class Solution {
public:
    vector<vector<int>> subsetsWithDup(vector<int>& nums) {
        sort(nums.begin(), nums.end());
        vector<vector<int>> ans = {{}};  // 一開始只有空集合
        int lastStart = 0;               // 上一輪「新增」的子集，在 ans 中的起始索引
        int n = nums.size();
        for (int i = 0; i < n; i++) {
            bool isDuplicate = (i > 0 && nums[i] == nums[i - 1]);
            // 不重複：替所有現有子集加上 nums[i]
            // 重複：只替「上一輪新增的那批」加上 nums[i]，
            //       因為更早的子集加上這個數字，在上一輪已經產生過了
            int from = isDuplicate ? lastStart : 0;
            int to = ans.size();  // 本輪開始前的大小，也是本輪新增子集的起點
            for (int j = from; j < to; j++) {
                vector<int> next = ans[j];  // 複製後再加，避免 push_back 擴容影響參考
                next.push_back(nums[i]);
                ans.push_back(next);
            }
            lastStart = to;  // 記下本輪新增的起點，供下一輪判斷
        }
        return ans;
    }
};
```

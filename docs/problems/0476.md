// 題目：476. Number Complement
// 難度：Easy
// 時間複雜度：O(1)
// 空間複雜度：O(1)
// 分類主題：bit-manipulation / math
// 技巧：用「位數相同的全 1 遮罩」減去 num，等同每個 bit 翻轉（補數）
//       bit_width 取得最高位 1 的位置，避免用迴圈逐位找

class Solution {
public:
    int findComplement(int num) {
        int bits = bit_width(static_cast<unsigned int>(num));
        long mask = (1u << bits) - 1;
        return static_cast<int>(mask - num);
    }
};

```cpp
// 題目：476. Number Complement
// 難度：Easy
// 時間複雜度：O(1)
// 空間複雜度：O(1)
// 分類主題：bit-manipulation / math
// 技巧：__builtin_clz 計算前導 0 個數 -> 得到位數 -> 全 1 遮罩減 num 即補數

class Solution {
public:
    int findComplement(int num) {
        // __builtin_clz(x)（count leading zeros）：32 位元下 x 的前導 0 個數（GCC／Clang 內建）
        // 位數 = 32 - 前導 0 個數
        // 例：num = 5（00...0101）：前導 0 有 29 個 -> bits = 3
        // 注意 num 不可為 0（clz(0) 未定義），題目保證 num >= 1
        int bits = 32 - __builtin_clz(num);

        // 建立 bits 個 1 的遮罩：(1 << bits) - 1
        // 例：bits = 3 -> 1000 - 1 = 111
        // 用 1L（64 位元）做位移：num 最大 2^31 - 1 時 bits = 31，
        // 若用 int 的 1 << 31 會溢位（C++20 前是未定義行為）
        int mask = (1u << bits) - 1;

        // 全 1 減去 num 等於把 num 每個 bit 翻轉（不會借位）
        // 例：111 - 101 = 010；等價於 mask ^ num
        // 結果一定落在 int 範圍內，轉回 int 回傳
        return static_cast<int>(mask - num);
    }
};
```

```cpp
/**
 * 題目：476. Number Complement (數字的補數)
 * 難度：簡單 (Easy)
 * 描述：給定一個正整數，將其二進位表示中的所有位元反轉（不包含前導零）。
 *
 * 時間複雜度：O(logN)
 * 空間複雜度：O(1)
 *
 * 設計思路：
 * 1. 1. 建立一個與 num 等長的 Mask，其位元皆為 1。
 * 2. 2. num ^ Mask 即為反轉後的結果。
 */

class Solution {
public:
    int findComplement(int num) {
        int mask = 0;
        int temp = num;
        while(temp > 0) {
            mask = (mask << 1) | 1;
            temp >>= 1;
        }
        return num ^ mask;
    }
};
```

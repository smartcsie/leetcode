```cpp
/**
 * 題目：190. Reverse Bits
 * 難度：簡單 (Easy)
 * 描述：反轉一個 32 位元無號整數的所有位元。
 *
 * 時間複雜度：O(1)
 * 空間複雜度：O(1)
 *
 * 解法思路：
 * 位元遍歷與拼湊
 * 1. 初始化結果 res = 0。
 * 2. 由於輸入是固定的 32 位元無符號整數，我們循環 32 次。
 * 3. 在每一次循環中：
 * - 將 res 向左移動一位 (res << 1)，為新的位元騰出空間。
 * - 取出 n 的最後一位 (n & 1)，並加到 res 的最低位。
 * - 將 n 向右移動一位 (n >> 1)，處理下一個位元。
 * 4. 循環結束後，res 即為反轉後的結果。
 */

class Solution {
public:
    int reverseBits(int n) {
        int m = 0;
        for(int i = 0; i < 32; i++) {
            m = (m << 1) | (n & 1);
            n >>= 1;
        }
        return m;
    }
};
```

```cpp
// 題目：190. Reverse Bits
// 難度：Easy
// 時間複雜度：O(1)（固定 5 步，與位元數 log 32 成正比）
// 空間複雜度：O(1)
// 分類主題：bit-manipulation
// 技巧：分治交換（divide and conquer）：依序交換 16 / 8 / 4 / 2 / 1 位元的區塊
//       概念：反轉 32 bits = 先左右半邊互換，再各半邊內部反轉，遞迴到每 1 bit

class Solution {
public:
    int reverseBits(int n) {
        // 第 1 步：高 16 位元與低 16 位元互換
        // 例：[AAAA BBBB] -> [BBBB AAAA]（每個字母代表 8 bits）
        n = (n >> 16) | (n << 16);

        // 第 2 步：每 16 位元內，高 8 位元與低 8 位元互換
        // 0xff00ff00 取出每組的高 8 位元，右移 8 放到低位
        // 0x00ff00ff 取出每組的低 8 位元，左移 8 放到高位
        n = ((n & 0xff00ff00) >> 8) | ((n & 0x00ff00ff) << 8);

        // 第 3 步：每 8 位元內，高 4 位元（nibble）與低 4 位元互換
        // 0xf0f0f0f0 = 1111 0000 ... ，0x0f0f0f0f = 0000 1111 ...
        n = ((n & 0xf0f0f0f0) >> 4) | ((n & 0x0f0f0f0f) << 4);

        // 第 4 步：每 4 位元內，高 2 位元與低 2 位元互換
        // 0xcccccccc = 1100 1100 ... ，0x33333333 = 0011 0011 ...
        n = ((n & 0xcccccccc) >> 2) | ((n & 0x33333333) << 2);

        // 第 5 步：每 2 位元內，相鄰兩個 bit 互換
        // 0xaaaaaaaa = 1010 1010 ... ，0x55555555 = 0101 0101 ...
        n = ((n & 0xaaaaaaaa) >> 1) | ((n & 0x55555555) << 1);

        // 經過 5 層交換（16、8、4、2、1），32 位元已完全反轉
        return n;
    }
};
```

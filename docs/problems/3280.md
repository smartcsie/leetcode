```cpp
/*
 * 題目：3280. Convert Date to Binary
 * 難度：Easy
 * 描述：給定格式為 yyyy-mm-dd 的日期字串 date，把年、月、日各自轉成二進位（不含前導零），並用 - 連接後回傳。
 * 分類主題：string
 * 技巧：substr 擷取固定位置, stoi 轉整數, std::format 的 b 格式轉二進位 (C++20)
 * 時間複雜度：O(1)（日期長度固定為 10）
 * 空間複雜度：O(1)
 * 解法思路：
 *   1. 日期格式固定為 yyyy-mm-dd，年在 [0, 4)、月在 [5, 7)、日在 [8, 10)
 *   2. 用 substr 取出三段，再用 stoi 轉成整數（stoi 會自動忽略前導零，如 "03" 轉為 3）
 *   3. std::format 的 {:b} 會輸出不含前導零的二進位，直接符合題目要求
 *   4. 三段用 - 串接即為答案
 */
class Solution {
public:
    string convertDateToBinary(string date) {
        // 格式固定為 yyyy-mm-dd：年 4 碼、月 2 碼、日 2 碼，中間各隔一個 -
        int year  = stoi(date.substr(0, 4));
        int month = stoi(date.substr(5, 2));
        int day   = stoi(date.substr(8, 2));
        // {:b} = 二進位，不含前導零；需要 C++20 的 <format>
        return format("{:b}-{:b}-{:b}", year, month, day);
    }
};
```

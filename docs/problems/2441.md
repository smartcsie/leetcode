```cpp
// 題目：2441. Largest Positive Integer That Exists With Its Negative
// 難度：Easy
// 時間複雜度：O(n)
// 空間複雜度：O(1)（固定大小 1001 的陣列，與 n 無關）
// 分類主題：array / hash-table
// 技巧：bitmask 標記（用 1 表示見過正數，2 表示見過負數）

class Solution {
public:
    int findMaxK(vector<int>& nums) {
        // count[v] 記錄絕對值為 v 的數字出現過哪些符號：
        //   bit 0 (值 1) = 出現過 +v
        //   bit 1 (值 2) = 出現過 -v
        //   值 3 = 正負都出現過
        // 題目保證 -1000 <= nums[i] <= 1000 且 nums[i] != 0，所以開 1001 格就夠
        // {} 讓所有元素初始化為 0
        array<uint8_t, 1001> count{};

        // 沒有任何數字同時有正負時，題目要求回傳 -1
        int mx = -1;

        for(const int& x : nums) {
            int idx = abs(x);                      // 以絕對值當索引

            // 正數標 1，負數標 2；用 |= 累積，重複出現同符號也不會影響結果
            count[idx] |= (x > 0) ? 1 : 2;

            // 兩種符號都見過（1 | 2 == 3），且比目前答案大就更新
            // 一旦 == 3，無論 +v 或 -v 是後出現的，都會在第二個出現時當下判斷到
            if(count[idx] == 3 && idx > mx) mx = idx;
        }
        return mx;
    }
};
```

## 解法：two-pointers-swap

**難度:** 🟢 E　**標籤:** Two Pointers Swap　**時間:** O(N)　**空間:** O(N)　**熟悉度:** 🟢 熟練

```cpp
/**
 * 題目：2441. Largest Positive Integer That Exists With Its Negative (與對應負數同時存在的最大正整數)
 * 難度：簡單 (Easy)
 * 描述：找出陣列中最大的正整數 k，使得 k 與 -k 同時存在於陣列中。
 *
 * 時間複雜度：O(NlogN)
 * 空間複雜度：O(1)
 *
 * 解法思路：
 * 1. 先對陣列進行升序排序，這樣負數會集中在左側，正數集中在右側。
 * 2. 使用雙指標：左指標 left 從最小值開始，右指標 right 從最大值開始。
 * 3. 比較 -nums[left] 與 nums[right] 的大小：
 * - 若相等：找到符合條件的最大正整數，直接回傳。
 * - 若 -nums[left] 較大：說明左側負數的絕對值太大，left 往右移尋找較小的負數。
 * - 若 nums[right] 較大：說明右側正數太大，沒有對應的負數，right 往左移尋找較小的正數。
 */

class Solution {
public:
    int findMaxK(vector<int>& nums) {
        int res = -1;
        sort(nums.begin(), nums.end());
        int left = 0;
        int right = nums.size() -1;
        while(left < right) {
            if(-nums[left] == nums[right] ) return nums[right];
            else if (-nums[left] > nums[right] ) left++;
            else right--;
        }
        return res;
    }
};
```

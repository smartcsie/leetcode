```cpp
/**
 * 題目：997. Find the Town Judge
 * 難度：Easy
 * 描述：小鎮有 n 個人，編號 1 ~ n，其中可能有一位法官。法官需滿足：
 *       1. 法官不信任任何人。
 *       2. 其他所有人都信任法官。
 *       trust[i] = [a, b] 表示 a 信任 b。
 *       若法官存在且唯一，回傳其編號，否則回傳 -1。
 *
 * 分類主題：graph-degree
 * 技巧：入度減出度, 單一陣列記錄淨信任分數, 分數恰為 n - 1 才是法官
 *
 * 時間複雜度：O(n + T)，T 為 trust 的長度
 * 空間複雜度：O(n)
 *
 * 解法思路：淨信任分數（入度 - 出度）
 * 1. 把信任關係看成有向邊 a → b。法官的入度是 n - 1（除了自己，每個人都信任他），
 *    出度是 0（不信任任何人）。
 * 2. 用一個陣列 counts 同時記錄兩件事：
 *    - 信任別人的人（邊的起點）：counts[a]--
 *    - 被信任的人（邊的終點）：counts[b]++
 *    所以 counts[i] = 入度 - 出度。
 * 3. 為什麼分數 n - 1 就足以判定法官：
 *    入度最多 n - 1（不含自己），出度最少 0，
 *    所以入度 - 出度 <= n - 1，等號成立若且唯若「入度 = n - 1 且出度 = 0」，
 *    剛好就是法官的兩個條件。
 * 4. 掃過 1 ~ n 找到分數為 n - 1 的人就回傳；找不到回傳 -1。
 *    符合條件的人最多只有一個（有人不被信任，入度就湊不滿 n - 1）。
 * 5. 邊界：n = 1 且 trust 為空時，counts[1] = 0 = n - 1，回傳 1，結果正確。
 */

class Solution {
public:
    int findJudge(int n, vector<vector<int>>& trust) {
        vector<int> counts(n + 1, 0);
        for(const auto& v : trust) {
            counts[v.front()]--;
            counts[v.back()]++;
        }
        for(int i = 1; i <= n; i++) {
            if(counts[i] == n - 1) return i;
        }
        return -1;
    }
};
```

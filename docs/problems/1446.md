```cpp
/**
 * 題目：1446. Consecutive Characters (連續字元)
 * 難度：簡單 (Easy)
 * 描述：找出字串中最長連續相同字元的長度。
 *
 * 時間複雜度：O(N)
 * 空間複雜度：O(1)
 *
 * 解法思路：
 * 1. 簡潔邏輯：移除額外的邊界判斷，讓主迴圈自動處理所有情況。
 * 2. 效能優化：減少每次迴圈呼叫 std::max 的次數，只有在字元改變或結束時才更新最大值。
 */

class Solution {
public:
    int maxPower(string s) {
        int count = 1;
        int mx = 0;
        for(int i = 0; i < s.size(); i++) {
            if(i == 0 || s[i] != s[i - 1]) count = 1;
            else count++;
            mx = max(mx, count);
        }
        return mx;
    }
};
```

```cpp
/**
 * 題目：2855. Minimum Right Shifts to Sort the Array
 * 難度：簡單 (Easy)
 * 描述：給定一個陣列，每次操作可以將最後一個元素移到最前面（右移），
 * 求最少幾次右移可以使陣列變成嚴格遞增排序，若不可能則回傳 -1。
 *
 * 時間複雜度：O(N)
 * 空間複雜度：O(1)
 *
 * 解法思路：
 * （找唯一的降序點）：
 * 1. 遍歷陣列，找出 nums[i] > nums[i+1] 的降序點。
 * 2. 合法的右移排序陣列只能有一個降序點。
 *    - 0 個降序點：已排序，回傳 0。
 *    - 1 個降序點：在 index i 處，右移次數為 n - (i + 1)。
 *    - 2 個以上降序點：無法透過右移排序，回傳 -1。
 * 3. 用 % n 處理環狀比較（最後一個元素與第一個元素的大小關係）。
 */
class Solution {
public:
    int minimumRightShifts(vector<int>& nums) {
        int count = 0;
        int n = nums.size();
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (nums[i] > nums[(i + 1) % n]) {
                ans = n - (i + 1);  // 降序點在 i，右移 n-(i+1) 次
                count++;
            }
            if (count == 2) return -1;  // 超過一個降序點，不可能排序
        }
        return ans;
    }
};
```

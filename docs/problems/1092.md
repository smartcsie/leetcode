# 1092. Shortest Common Supersequence

```cpp
/**
 * 題目：1092. Shortest Common Supersequence
 * 難度：困難 (Hard)
 * 描述：給定兩個字串 str1 和 str2，找出最短的字串使得
 * str1 和 str2 都是它的子序列，回傳該字串。
 *
 * 時間複雜度：O(N * M)
 * 空間複雜度：O(N * M)
 *
 * 解法思路：
 * （LCS + 回溯建構結果字串）：
 * 1. 先求 str1 和 str2 的 LCS（最長公共子序列）。
 *    LCS 的部分只需出現一次，非 LCS 的部分各自保留。
 *    最短公共超序列長度 = n + m - LCS(str1, str2)。
 * 2. 從 dp 表右下角往左上角回溯建構結果字串：
 *    - str1[i-1] == str2[j-1]（LCS 字元）：加一次，i-- j--
 *    - dp[i-1][j] > dp[i][j-1]（來自 str1）：加 str1[i-1]，i--
 *    - 否則（來自 str2）：加 str2[j-1]，j--
 * 3. 處理剩餘未加入的字元。
 * 4. 因為是從右下角回溯，最後需要 reverse。
 */
class Solution {
public:
    string shortestCommonSupersequence(string str1, string str2) {
        int n = str1.size(), m = str2.size();
        vector<vector<int>> dp(n + 1, vector<int>(m + 1, 0));

        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= m; j++)
                if (str1[i-1] == str2[j-1])
                    dp[i][j] = 1 + dp[i-1][j-1];
                else
                    dp[i][j] = max(dp[i-1][j], dp[i][j-1]);

        // 回溯建構結果字串
        string res = "";
        int i = n, j = m;
        while (i > 0 && j > 0) {
            if (str1[i-1] == str2[j-1]) {
                res += str1[i-1];  // LCS 字元，只加一次
                i--; j--;
            } else if (dp[i-1][j] > dp[i][j-1]) {
                res += str1[i-1];  // 來自 str1
                i--;
            } else {
                res += str2[j-1];  // 來自 str2
                j--;
            }
        }
        while (i > 0) { res += str1[i-1]; i--; }
        while (j > 0) { res += str2[j-1]; j--; }

        reverse(res.begin(), res.end());
        return res;
    }
};
```

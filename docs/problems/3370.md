```cpp
/**
 * 題目：3370. Smallest Number With All Set Bits
 * 難度：簡單 (Easy)
 * 描述：給定一個正整數 n，找出最小的整數使得其二進位表示中所有位元都為 1，
 * 且該整數大於或等於 n。
 *
 * 時間複雜度：O(logN)
 * 空間複雜度：O(1)
 *
 * 解法思路：
 * （bit_width + 全 1 遮罩）：
 * 1. 用 bit_width 計算 n 需要幾個 bit 表示。
 * 2. 建立一個相同位數、所有 bit 都為 1 的數字。
 *    例如 n=5（101），bit_width=3，建立 111=7。
 * 3. 全 1 的數字一定 >= n，且是最小的滿足條件的數字。
 *
 * 等價寫法：(1 << bits) - 1
 */
class Solution {
public:
    int smallestNumber(int n) {
        int bits = bit_width(static_cast<unsigned int>(n));
        return (1u << bits) - 1;
    }
};
```

```cpp
// 題目：3370. Smallest Number With All Set Bits
// 難度：Easy
// 時間複雜度：O(1)
// 空間複雜度：O(1)
// 分類主題：bit-manipulation / math
// 技巧：用 log2 算出 n 的位數，答案是「位數相同的全 1 數字」(2^bits - 1)

class Solution {
public:
    int smallestNumber(int n) {
        // 找出 >= n 的最小「二進位全是 1」的數字（1、3、7、15、31...）。
        // 這種數字一定是 2^k - 1；而 n 的位數就是能容納它的最小 k。
        // 位數 = floor(log2(n)) + 1
        // 例：n = 5（101）：log2(5) ≈ 2.32 -> 2 + 1 = 3 位
        //     n = 8（1000）：log2(8) = 3 -> 3 + 1 = 4 位
        // 指定給 int 時小數會被捨去（相當於 floor），題目保證 n >= 1，log2 才有意義
        int bits = log2(n) + 1;

        // bits 個 1：(1 << bits) - 1
        // 例：bits = 3 -> 1000 - 1 = 111 = 7（>= 5）
        // 如果 n 本身就是全 1（如 7），bits 仍是 3，回傳 7 自己，符合「>= n」
        // n <= 1000，bits 最多 10，不會溢位
        return (1u << bits) - 1;
    }
};
```

```cpp
/**
 * 題目：3370. Smallest Number With All Set Bits (具有所有位元為 1 的最小數字)
 * 難度：簡單 (Easy)
 * 描述：回傳一個大於或等於 n 的最小整數，其二進位表示皆為 1。
 *
 * 時間複雜度：O(logN)
 * 空間複雜度：O(1)
 *
 * 解法思路：
 * 1. 邏輯簡化：利用 `(1 << bit_count) - 1` 的性質，這比逐位元位移更具可讀性。
 */

class Solution {
public:
    int smallestNumber(int n) {
        int mask = 0;
        while(n > 0) {
            mask = mask << 1 | 1;
            n >>= 1;
        }
        return mask;
    }
};
```

```cpp
// 題目：3370. Smallest Number With All Set Bits
// 難度：Easy
// 時間複雜度：O(1)
// 空間複雜度：O(1)
// 分類主題：bit-manipulation / math
// 技巧：__builtin_clz 求出 n 的位數，答案是位數相同的全 1 數字 (2^bits - 1)

class Solution {
public:
    int smallestNumber(int n) {
        // 找出 >= n 的最小「二進位全是 1」的數字（1、3、7、15、31...）。
        // 這種數字一定是 2^k - 1，k 是 n 的二進位位數。
        //
        // __builtin_clz(x)（count leading zeros）：32 位元下 x 的前導 0 個數
        // （GCC／Clang 內建；x 不可為 0，題目保證 n >= 1）
        // 位數 = 32 - 前導 0 個數
        // 例：n = 5（00...0101）：前導 0 有 29 個 -> cnt = 3
        int cnt = 32 - __builtin_clz(n);

        // cnt 個 1：(1 << cnt) - 1
        // 例：cnt = 3 -> 1000 - 1 = 111 = 7（>= 5）
        // 如果 n 本身就是全 1（如 7），cnt 仍是 3，回傳 7 自己，符合「>= n」
        // 這題 n <= 1000，cnt 最多 10，不會溢位
        return (1u << cnt) - 1;
    }
};
```

```cpp
/**
 * 題目：129. Sum Root to Leaf Numbers (求根節點到葉節點數字之和)
 * 難度：中等 (Medium)
 * 描述：計算二元樹中所有從根節點到葉節點的路徑所代表的數字之和。
 *
 * 時間複雜度：O(N)
 * 空間複雜度：O(H)
 *
 * 解法思路：
 * 1. 使用 DFS 遍歷，每個節點將「當前值 * 10 + 節點值」傳遞給子節點。
 * 2. 傳值 (Pass-by-value) 傳遞 `curSum`，這能自動處理路徑狀態，不需要手動回溯。
 */

/**
 * Definition for a binary tree node.
 * struct TreeNode {
 * int val;
 * TreeNode *left;
 * TreeNode *right;
 * TreeNode() : val(0), left(nullptr), right(nullptr) {}
 * TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 * TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */

class Solution {
private:
    void dfs(TreeNode* root, long cur, long& sum) {
        if(!root) return;
        cur = 10 * cur + root->val;
        if(!root->left && !root->right) {
            sum += cur;
            return;
        }
        if(root->left) dfs(root->left, cur, sum);
        if(root->right) dfs(root->right, cur, sum);

    }
public:
    int sumNumbers(TreeNode* root) {
        long sum = 0;
        dfs(root, 0, sum);
        return static_cast<int>(sum);
    }
};
```

## 解法：tree-dfs-path

**難度:** 🟡 M　**標籤:** Tree Dfs Path　**時間:** O(n)　**空間:** O(1)　**熟悉度:** 🟣 易忘　👑 **代表題（DFS Early Return）**　**官方解答:** ✅ 已同步官方解答

```cpp
/*
 * 題目：129. Sum Root to Leaf Numbers
 * 難度：Medium
 * 描述：二元樹每個節點存放 0 到 9 的數字，每條從根到葉的路徑代表一個數字（例如 1 → 2 → 3 代表 123）。回傳所有根到葉路徑所代表數字的總和。
 * 分類主題：tree-traversal
 * 技巧：Morris 走訪（線索二元樹）, 用 steps 回溯數字位數, 以 pred 的左子樹判斷葉節點
 * 時間複雜度：O(n)
 * 空間複雜度：O(1)
 * 解法思路：
 *   1. 一般解法是遞迴 DFS，把 cur × 10 + val 往下傳，到葉節點加總；這裡改用 Morris 走訪，不用遞迴堆疊，空間降到 O(1)
 *   2. Morris 走訪：對有左子樹的 root，找左子樹中最右的節點 pred（中序前驅），暫時讓 pred 的右指標指回 root 當作線索
 *   3. 第一次到 root：把 root 的值接到 cur 尾端，建立線索，往左子樹走
 *   4. 透過線索第二次回到 root：左子樹已走完；若 pred 沒有左子節點，pred 是葉節點，這條路徑完成，把 cur 加進總和
 *   5. 回到 root 時要把 cur 還原：左子樹從 root->left 到 pred 的那條鏈共 steps 個節點，各接上一位數字，所以除以 10 共 steps 次，之後再拆掉線索並改走右子樹
 *   6. 沒有左子樹的節點：直接接上數字；若也沒有右子樹，它是葉節點，把 cur 加進總和；再往右走（可能是真的右子節點，也可能是線索回到祖先）
 */
class Solution {
public:
    int sumNumbers(TreeNode* root) {
        int total = 0;   // 所有「根到葉」數字的總和
        int cur = 0;     // 目前從根走到當前位置所組成的數字

        while(root) {
            if(root->left) {
                // 找 root 在中序下的前驅：root 左子樹中最右邊的節點
                TreeNode* pred = root->left;
                int steps = 1;   // 從 root->left 到 pred 經過的節點數（等於 cur 之後多出的位數）
                while(pred->right && pred->right != root) {
                    pred = pred->right;
                    steps++;
                }

                if(!pred->right) {
                    // 第一次到 root：先把 root 的數字接上，建立線索，再往左子樹走
                    cur = cur * 10 + root->val;
                    pred->right = root;
                    root = root->left;
                } else {
                    // 第二次回到 root（透過線索）：左子樹已走完
                    if(!pred->left) total += cur;           // pred 是葉節點，這條路徑完成
                    for(int i = 0; i < steps; i++) cur /= 10;   // 退掉左子樹那條鏈加上的位數
                    pred->right = nullptr;                  // 拆掉線索，還原樹的結構
                    root = root->right;                     // 換走右子樹
                }
            } else {
                // 沒有左子樹：直接接上數字
                cur = cur * 10 + root->val;
                if(!root->right) total += cur;   // 沒有右子樹也是葉節點，這條路徑完成
                root = root->right;              // 可能是真正的右子節點，也可能是線索回到祖先
            }
        }
        return total;
    }
};
```

> **ans不等於nullptr，表示找到答案**<br>

```cpp
/**
 * 題目：1038. Binary Search Tree to Greater Sum Tree (將 BST 轉為大於值總和樹)
 * 難度：中等 (Medium)
 * 描述：將 BST 的每個節點值替換為原節點值加上所有大於該節點值的節點之和。
 *
 * 時間複雜度：O(N)
 * 空間複雜度：O(H)
 *
 * 解法思路：
 * 1. 反向中序遍歷：按照「右、根、左」的順序遍歷，數值會由大到小累積。
 * 2. 累加狀態：使用變數 `sum` 維護當前遍歷過所有節點的累計總和。
 */

class Solution {
private:
    void reverseInorder(TreeNode* root, int& pre) {
        if(!root) return;
        reverseInorder(root->right, pre);
        if(pre != -1) root->val += pre;
        pre = root->val;
        reverseInorder(root->left, pre);
    }
public:
    TreeNode* bstToGst(TreeNode* root) {
        int dummy = -1;
        reverseInorder(root, dummy);
        return root;
    }
};
```

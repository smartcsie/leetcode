```cpp
/**
 * 題目：692. Top K Frequent Words (前 K 個高頻單詞)
 * 難度：中等 (Medium)
 * 描述：找出陣列中出現頻率前 K 高的單詞，若頻率相同則按字典序排序。
 *
 * 時間複雜度：O(NlogK)
 * 空間複雜度：O(N)
 *
 * 解法思路：
 * 當 K << N 時，使用 Heap (優先佇列) 比全排序 (sort) 的 O(N log N) 更有效率。
 */

class Solution {
public:
    vector<string> topKFrequent(vector<string>& words, int k) {
        unordered_map<string, int> map;
        for(const string& s : words) map[s]++;
        auto comp = [](const auto& a, const auto& b) {
            if (a.first == b.first) return a.second > b.second; 
            return a.first < b.first;        
        };
        vector<pair<int, string>> items;
        items.reserve(map.size());
        for(const auto& [s, count] : map) items.push_back({count, s});
        priority_queue<pair<int, string>, vector<pair<int, string>>, decltype(comp)> pq(comp, move(items));
        vector<string> ans(k);
        for(int i = 0; i < k; i++) {
            ans[i] = pq.top().second;
            pq.pop();
        }
        return ans;
    }
};
```

/**
 * 題目：1935. Maximum Number of Words You Can Type
 * 難度：簡單 (Easy)
 * 描述：計算在給定壞掉鍵盤的情況下，有多少個完整的單字可以被完全輸入。
 *
 * 時間複雜度：O(N*M)
 * 空間複雜度：O(1)
 *
 * 解法思路：
 * 1. 使用 istringstream 將 text 切分成多個單字。
 * 2. 針對每個單字進行檢查，若單字中包含任何一個 brokenLetters，則該單字無法輸入。
 * 3. 統計未被標記為「損壞」的單字數量。
 */

class Solution {
public:
    int canBeTypedWords(string text, string brokenLetters) {
        bitset<26> broken;
        for(const char& c : brokenLetters) broken.set(c - 'a');
        int n =  text.size();
        bool pass = true;
        int canType = 0;
        for(int i = 0, start = 0; i <= n; i++) {
            if(islower(text[i]) && broken.test(text[i] - 'a')) pass = false;
            if(i == n || text[i] == ' ') {
                if(pass) canType++;
                pass = true;
            }
        }
        return canType;
    }
};

```cpp
/**
 * 題目：3438. Find Valid Pair of Adjacent Digits in String
 * 難度：簡單 (Easy)
 * 描述：在字串中尋找滿足以下條件的第一個相鄰數對：
 * 1. 兩個數字不相等。
 * 2. 每個數字在字串中的出現頻率等於該數字本身。
 *
 * 時間複雜度：O(N)
 * 空間複雜度：O(1)
 */

class Solution {
public:
    string findValidPair(string s) {
        vector<int> count(10, 0);
        for(const char& c : s) count[c - '0']++;
        for(int i = 0; i < s.size() - 1; i++) {
            int k1 = s[i] - '0';
            int k2 = s[i + 1] - '0';
            if(k1 != k2  && count[k1] == k1 && count[k2] == k2)
                return string(1, s[i]) + string(1, s[i + 1]);
        }
        return "";
    }
};
```

/**
 * 題目：1185. Day of the Week
 * 難度：中等 (Medium)
 * 描述：給你一個日期，請你設計一個演算法返回對應的星期幾（英文名稱）。
 * 範圍通常在 1971 年到 2100 年之間。
 *
 * 時間複雜度：O(1)
 * 空間複雜度：O(1)
 *
 * 解法思路：
 * （蔡勒公式 Zeller's Congruence）：
 * 1. 蔡勒公式：
 * - 是一個由數學家克里斯蒂安·蔡勒推導出來的計算公曆（格里曆）任何一天是星期幾的公式。
 * 2. 變數定義與調整：
 * - 若月份為 1 月或 2 月，則被視為上一年的 13 月和 14 月，因此年份 `year` 需減 1，`month` 需加 12。
 * - `year1` 代表世紀數（`year / 100`），`year2` 代表世紀內的年份（`year % 100`）。
 * 3. 公式計算：
 * - $W = (year2 + \lfloor year2 / 4 \rfloor + \lfloor year1 / 4 \rfloor - 2 \cdot year1 + \lfloor 13(month + 1) / 5 \rfloor + day - 1) \pmod 7$
 * - 透過處理餘數可能為負數的情況（`+7`），對應到正確的星期字串陣列。
 */

class Solution {
public:
    std::string dayOfTheWeek(int day, int month, int year) {
        // 定義星期字串陣列（對應標準蔡勒公式的餘數索引）
        std::vector<std::string> days = {"Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"};
        // 蔡勒公式規則：1 月和 2 月要算作上一年的 13 月和 14 月
        if (month <= 2) {
            month += 12;
            year -= 1;
        }
        int year1 = year / 100; // 世紀
        int year2 = year % 100; // 年份後兩位
        // 標準蔡勒公式
        int dayIdx = (year2 + year2 / 4 + year1 / 4 - 2 * year1 + (13 * (month + 1)) / 5 + day - 1) % 7;
        // 處理 C++ 中負數取模可能小於 0 的情況
        if (dayIdx < 0) {
            dayIdx += 7;
        }
        return days[dayIdx];
    }
};
```

/**
 * 題目：1185. Day of the Week
 * 難度：Easy
 * 時間複雜度：O(1)
 * 空間複雜度：O(1)
 * 分類主題：math-formula
 * 技巧：Sakamoto's algorithm
 *
 * 解法說明：
 * 使用 Sakamoto's algorithm（Zeller's congruence 的簡化版）直接用公式算出星期幾：
 *   星期 = (y + y/4 - y/100 + y/400 + t[m-1] + d) mod 7，結果 0 = 星期日
 *   - y + y/4 - y/100 + y/400：從西元元年累積的天數，除以 7 取餘數
 *     （每年 365 天，365 mod 7 = 1，所以每年星期往後推 1 天，再加上閏年多出的天數）
 *   - t[m-1]：該月 1 日相對於 1 月 1 日的星期偏移（把前面各月的天數 mod 7 預先算好）
 *   - d：當月第幾天
 * 閏日（2/29）在年底計算會比較方便，所以 1、2 月要當成「前一年」的 13、14 月。
 */
class Solution {
    // 索引 0 = 星期日，順序要與公式的取餘結果一致
    static constexpr array<const char*, 7> days = {"Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"};

    // 各月份的星期偏移表（Sakamoto's table）：
    // 前面各月天數累加後 mod 7 的結果，並已配合「1、2 月視為前一年」做過調整
    static constexpr array<int, 12> dayIdx = {0, 3, 2, 5, 0, 3, 5, 1, 4, 6, 2, 4};
public:
    string dayOfTheWeek(int day, int month, int year) {
        // 1、2 月當成前一年，這樣閏年多出的 2/29 會落在「這一年」的最後，
        // 公式中 year/4 - year/100 + year/400 才能正確算入該年的閏日
        if (month < 3) year -= 1;

        // 公式各項加總後 mod 7，得到星期索引
        int idx = (year + year / 4 - year / 100 + year / 400 + dayIdx[month - 1] + day) % 7;
        return days[idx];
    }
};
```

```cpp
/**
 * 題目：566. Reshape the Matrix
 * 難度：Easy
 * 描述：給定 m x n 的矩陣 mat，與整數 r、c，
 *       將 mat 依「列優先」(row-major) 的順序重塑成 r x c 的矩陣。
 *       若 m * n != r * c，無法重塑，回傳原矩陣。
 *
 * 分類主題：matrix-index-mapping
 * 技巧：二維座標與一維索引互轉, idx = i * n + j, 再用 idx / c、idx % c 還原新座標
 *
 * 時間複雜度：O(m * n)，每個元素搬移一次
 * 空間複雜度：O(r * c)（輸出矩陣，不計輸出為 O(1)）
 *
 * 解法思路：攤平成一維索引再重新切列
 * 1. 元素總數不同（m * n != r * c）就無法重塑，直接回傳原矩陣。
 *    尺寸完全相同（m == r 且 n == c）時結果也是原矩陣，一併提早回傳。
 * 2. 列優先的順序下，原矩陣 (i, j) 是第 idx = i * n + j 個元素
 *    （前面有 i 整列、每列 n 個，再加上本列的第 j 個）。
 * 3. 新矩陣每列有 c 個元素，所以第 idx 個元素落在：
 *    - 第 idx / c 列
 *    - 第 idx % c 行
 * 4. 逐一把 mat[i][j] 填到 ans[idx / c][idx % c]，就完成重塑。
 */
class Solution {
public:
    vector<vector<int>> matrixReshape(vector<vector<int>>& mat, int r, int c) {
        int m = mat.size();     // 原矩陣列數
        int n = mat[0].size();  // 原矩陣行數

        // 尺寸相同，不需要重塑
        if(m == r && n == c) return mat;

        // 元素總數不同，無法重塑，回傳原矩陣
        if(m * n != r * c) return mat;

        vector<vector<int>> ans(r, vector<int>(c, 0));

        for(int i = 0; i < m; i++) {
            for(int j = 0; j < n; j++) {
                int idx = i * n + j;  // (i, j) 在列優先順序下的一維編號
                int k = idx / c;      // 對應到新矩陣的列
                int l = idx % c;      // 對應到新矩陣的行
                ans[k][l] = mat[i][j];
            }
        }
        return ans;
    }
};
```

```cpp
/**
 * 題目：777. Swap Adjacent in LR String
 * 難度：中等 (Medium)
 * 描述：字串只含 'L'、'R'、'X'。每次操作可以把 "XL" 換成 "LX"，
 *       或把 "RX" 換成 "XR"。判斷 start 能否經過若干次操作變成 result。
 *
 * 分類主題：two-pointers-other
 * 技巧：雙指標跳過 X, 比較非 X 字元的相對順序, L 只能左移 R 只能右移
 *
 * 時間複雜度：O(N)，兩個指標各只往右走一次
 * 空間複雜度：O(1)
 *
 * 解法思路：雙指標比對非 X 字元
 * 1. 操作的效果：L 只能往左移（XL → LX），R 只能往右移（RX → XR），
 *    而且 L 與 R 之間無法互相穿越，所以去掉所有 X 後，
 *    start 與 result 的 L、R 序列必須完全相同。
 * 2. i 指向 start、j 指向 result，各自跳過 'X'，停在下一個非 X 字元。
 * 3. 兩邊同時走到尾端，代表所有字母都配對成功，回傳 true；
 *    只有一邊走完，代表字母數量不同，回傳 false。
 * 4. 配對到的兩個字母必須相同，否則順序不一致，回傳 false。
 * 5. 再檢查位置是否合法：
 *    - 'L' 只能左移，所以 start 中的位置 i 必須大於或等於 result 中的位置 j，
 *      若 i < j 代表需要右移，回傳 false。
 *    - 'R' 只能右移，所以 i 必須小於或等於 j，
 *      若 i > j 代表需要左移，回傳 false。
 * 6. 配對成功後兩個指標各前進一格，繼續比對下一組字母。
 */


class Solution {
public:
    bool canTransform(string start, string result) {
        int m = start.size();
        int n = result.size();  
        int i = 0,j = 0; 
        while(i < m|| j < n){ 
            // i走到第一個不是X
            while(i < m && start[i] == 'X') i++;
            // j走到第一個不是X
            while(j < n && result[j] == 'X') j++;
            //如果 i 和 j 同時走到了字串尾端，代表所有非 'X' 的字母都成功配對完畢
            if(i == m && j == n) return true;

            //如果其中一個先走到底，另一個還沒，代表兩邊剩餘的字母數量不對
            if(i == m || j == n) return false;

            // 字母順序不一樣，回傳false
            if (start[i] != result[j]) return false;

            // L出現，start的L往左移動才合法
            if(start[i]=='L' && i < j) return false; 

            // R出現，start的R往右移動才合法
            if(start[i]=='R' && i > j) return false; 
            i++;
            j++;
        } 
        return true;
    }
};
```

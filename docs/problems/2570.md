/**
 * 題目：2570. Merge Two 2D Arrays by Summing Values
 * 難度：Easy
 * 描述：nums1、nums2 的每個元素為 [id, val]，且各自以 id 嚴格遞增排列。
 *       合併兩個陣列：id 出現在任一陣列就保留，
 *       若兩邊都有則 val 相加，回傳依 id 遞增排列的結果。
 *
 * 分類主題：hash-table-counting
 * 技巧：值域固定的計數陣列, 依 id 掃描天然有序
 *
 * 解法思路：計數陣列
 * 1. id 的範圍只有 1 ~ 1000，直接開 counts[1001]，以 id 當索引累加 val。
 * 2. 兩個陣列都走一遍，同 id 的 val 自然加在同一格。
 * 3. 從 id = 1 掃到 1000，把 counts[i] 不為 0 的格子輸出為 [i, counts[i]]。
 *    掃描順序就是 id 遞增，不需要另外排序。
 * 4. 以「counts 是否為 0」判斷 id 是否存在，是因為題目保證 val >= 1。
 *
 * 時間複雜度：O(n + m + V)，V = 1000 為 id 值域
 * 空間複雜度：O(V)（不計輸出）
 */
class Solution {
public:
    vector<vector<int>> mergeArrays(vector<vector<int>>& nums1, vector<vector<int>>& nums2) {
        vector<vector<int>> ans;
        array<int, 1001> counts = {}; // counts[id] = 該 id 的 val 總和

        // 兩個陣列的 val 依 id 累加到同一張表
        for (const vector<int>& v : nums1) counts[v[0]] += v[1];
        for (const vector<int>& v : nums2) counts[v[0]] += v[1];

        // 依 id 由小到大掃描，val 總和為 0 代表該 id 沒出現過
        for (int i = 1; i <= 1000; i++) {
            if (!counts[i]) continue;
            ans.push_back({i, counts[i]});
        }
        return ans;
    }
};
```

## 解法：two-pointers-same

/*
 * 題目：2570. Merge Two 2D Arrays by Summing Values
 * 難度：Easy
 * 描述：給定兩個二維陣列 nums1、nums2，每個元素是 [id, val]，且各自已依 id 嚴格遞增排序、id 不重複。合併成一個依 id 遞增的陣列：兩邊都有的 id，值相加；只有一邊有的 id，值不變（視為另一邊為 0）。
 * 分類主題：two-pointers-same
 * 技巧：雙指標合併已排序陣列, 相同 id 相加, 收尾搬運剩餘元素
 * 時間複雜度：O(m + n)
 * 空間複雜度：O(1)（不含輸出）
 * 解法思路：
 *   1. 兩個陣列都已依 id 排序，像合併兩個有序串列一樣，用 i、j 各自指向目前元素
 *   2. id 相同：把兩邊的值相加，放入結果，兩個指標都往前
 *   3. nums1 的 id 較小：它在 nums2 中不存在，直接放入結果，i 往前
 *   4. nums2 的 id 較小：同理，直接放入結果，j 往前
 *   5. 其中一邊走完後，把另一邊剩下的元素依序接到結果後面（剩下的 id 一定都更大，且已排序）
 */
class Solution {
public:
    vector<vector<int>> mergeArrays(vector<vector<int>>& nums1, vector<vector<int>>& nums2) {
        int i = 0, j = 0;
        vector<vector<int>> ans;
        while((i < nums1.size()) && (j < nums2.size())) {
            if(nums1[i][0] == nums2[j][0]) {
                // 兩邊都有這個 id：值相加，兩個指標一起前進
                ans.push_back({nums1[i][0], nums1[i][1] + nums2[j][1]});
                i++;
                j++;
            } else if(nums1[i][0] < nums2[j][0]) {
                // 只有 nums1 有這個 id：原樣放入
                ans.push_back(nums1[i++]);
            } else {
                // 只有 nums2 有這個 id：原樣放入
                ans.push_back(nums2[j++]);
            }
        }
        // 其中一邊用完，另一邊剩下的 id 都更大，依序接上即可
        while(i < nums1.size()) ans.push_back(nums1[i++]);
        while(j < nums2.size()) ans.push_back(nums2[j++]);
        return ans;
    }
};
```

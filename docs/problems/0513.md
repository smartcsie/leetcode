# 513. Find Bottom Left Tree Value

```cpp
/**
 * 題目：513. Find Bottom Left Tree Value (找樹左下角的值)
 * 難度：中等 (Medium)
 * 描述：給定一個二元樹的根節點 root，找出該樹「最後一層」中「最左邊」的節點值。
 *
 * 時間複雜度：O(N)
 * 空間複雜度：O(W)
 *
 * 解法思路：
 * 1. 使用廣度優先搜索 (BFS / 層序遍歷)。
 * 2. 在進入每一層的 while 迴圈時，該層的第一個節點（即 q.front()）就是該層最左邊的節點。
 * 3. 我們不斷更新 res 為每一層的第一個節點值。
 * 4. 當遍歷結束後，res 儲存的自然就是「最後一層」的「第一個節點值」。
 */

class Solution {
public:
    int findBottomLeftValue(TreeNode* root) {
        if(!root) return 0;
        queue<TreeNode*> q({root});
        int leftmost = 0;
        while(!q.empty()) {
            leftmost = q.front()->val;
            for(int i = q.size() - 1; i >= 0; i--) {
                TreeNode* node = q.front();
                q.pop();
                if(node->left) q.push(node->left);
                if(node->right) q.push(node->right);
            }
        }
        return leftmost;
    }
};
```

## 解法：tree-dfs-path

```cpp
// 題目：513. Find Bottom Left Tree Value
// 難度：Medium
// 時間複雜度：O(n)
// 空間複雜度：O(h)，h = 樹高（遞迴堆疊；最壞退化成鏈時為 O(n)）
// 分類主題：tree / binary-tree
// 技巧：DFS 前序（先左後右）+ 深度記錄；只在「更深一層」時更新答案，所以每層只有最左節點會被記下

class Solution {
public:
    int findBottomLeftValue(TreeNode* root) {
        int maxDepth = -1;   // 目前見過的最深深度（-1 表示還沒走到任何節點）
        int ans = 0;         // 最深那一層最左邊的值
        dfs(root, 0, maxDepth, ans);   // 根節點的深度是 0
        return ans;                    // 題目保證樹至少有一個節點
    }

private:
    // maxDepth、ans 用參考傳遞，讓整個遞迴過程共用同一份狀態
    void dfs(TreeNode* node, int depth, int& maxDepth, int& ans) {
        if(!node) return;

        // 第一次走到更深的一層：
        // 因為先走左子樹，同一層裡最先被走到的節點一定是最左邊的那個，
        // 所以這個節點就是該層的最左節點。
        // 一定要用 > 而不是 >=，否則同一層後面走到的右邊節點會覆蓋掉左邊的答案
        if(depth > maxDepth) {
            maxDepth = depth;
            ans = node->val;
        }

        dfs(node->left, depth + 1, maxDepth, ans);    // 一定要先左
        dfs(node->right, depth + 1, maxDepth, ans);   // 再右
    }
};
```

/**
 * 題目：343. Integer Break
 * 難度：中等 (Medium)
 * 描述：給定一個整數 n，將其拆分為 k 個正整數之和 (k >= 2)，使得這些整數的乘積最大。
 *
 * 時間複雜度：O(1)
 * 空間複雜度：O(1)
 *
 * 解法思路：
 * (數學貪婪法)：
 * 1. 若 n <= 3，直接回傳 n - 1 (例如 n=2 為 1*1=1, n=3 為 2*1=2)。
 * 2. 對於 n > 3，儘量拆出 3。因為 3 是小於 5 的數中，乘積成長率最高的數。
 * 3. 當剩餘數字為 4 時，保留 4 (即 2 * 2) 比拆成 3 * 1 更大。
 * 4. 循環將 n 減去 3 並累乘，最後乘上剩下的數字。
 */

class Solution {
public:
    int integerBreak(int n) {
        // 基本邊界情況
        if (n <= 3) return n - 1;
        
        int product = 1;
        // 儘量將 n 拆成 3 的倍數
        while (n > 4) {
            product *= 3;
            n -= 3;
        }
        
        // 最後剩餘的 n 必定是 2, 3 或 4
        return product * n;
    }
};
```

/**
 * 題目：343. Integer Break
 * 難度：Medium
 * 描述：給定整數 n，將它拆成至少兩個正整數的和，
 *       回傳這些整數乘積的最大值。
 *
 * 分類主題：math-formula
 * 技巧：盡量拆成 3, 餘數 1 時改用 4 而不是 3 加 1, 餘數 2 直接乘 2, n 小於等於 3 要特判（至少要拆成兩份）
 *
 * 時間複雜度：O(log n)（pow 的指數運算，本題 n 很小，實務上可視為 O(1)）
 * 空間複雜度：O(1)
 *
 * 解法思路：數學 + 貪心，盡量拆出 3
 * 1. n 太小無法再拆得更好：題目要求至少拆成兩份，
 *    n = 2 只能 1 + 1，乘積 1；n = 3 最好是 1 + 2，乘積 2。
 *    兩者都等於 n - 1，所以 n <= 3 直接回傳 n - 1。
 * 2. 拆數字的原則：
 *    - 每份大於等於 5 時，拆成 2 與 (x - 2) 的乘積 2 × (x - 2) = 2x - 4，大於 x，
 *      所以最佳拆法裡不會有大於等於 5 的數。
 *    - 4 不用再拆，因為 2 × 2 = 4，拆或不拆乘積相同。
 *    - 1 不該出現（乘 1 等於浪費一份）。
 *    - 在 2 與 3 之間，3 比較划算：兩個 3 的和是 6、乘積 9，
 *      三個 2 的和也是 6、乘積卻只有 8，所以盡量用 3。
 * 3. 依 n 除以 3 的餘數分三種情況：
 *    - 餘 0：全部拆成 3，乘積 = 3 的 (n / 3) 次方。
 *    - 餘 1：多出一個 1，與其 3 + 1（乘積 3），不如把一個 3 與 1 合併成 4（拆成 2 + 2），
 *      乘積 = 3 的 (n / 3 - 1) 次方 × 4。
 *    - 餘 2：多出一個 2，直接乘上去，乘積 = 3 的 (n / 3) 次方 × 2。
 * 4. 餘 1 的情況要求 n / 3 >= 1，n >= 4 時成立（n <= 3 已在前面處理）。
 * 5. 溢位檢查：我印象中 n 最大約 58，最大乘積約 1.5 × 10^9，仍在 int 範圍內。
 */
class Solution {
public:
    int integerBreak(int n) {
        // 至少要拆成兩份：n = 2 → 1 + 1，n = 3 → 1 + 2
        if (n <= 3) {
            return n - 1;
        }
        // 全部拆成 3
        if (n % 3 == 0) {
            return pow(3, n / 3);
        }
        // 多出一個 1：把一個 3 與這個 1 合併成 4（2 + 2），不要留 3 + 1
        if (n % 3 == 1) {
            return pow(3, n / 3 - 1) * 4;
        }
        // 多出一個 2：直接乘上去
        return pow(3, n / 3) * 2;
    }
};
```

```cpp
// 題目：1748. Sum of Unique Elements
// 難度：Easy
// 時間複雜度：O(n)
// 空間複雜度：O(1)（固定 101 格的計數陣列）
// 分類主題：array / hash-table
// 技巧：計數陣列（值域小）、一次掃描、依出現次數即時加減

class Solution {
public:
    int sumOfUnique(vector<int>& nums) {
        // 題目限制 1 <= nums[i] <= 100，直接用數值當索引
        // counts[v] = v 目前出現的次數；最多 100 次，uint8_t（0~255）足夠
        array<uint8_t, 101> counts{};
        int sum = 0;   // 目前「只出現一次」的數字總和
        for(const int& x : nums) {
            counts[x]++;
            // 第 1 次出現：x 目前是唯一的，加進總和
            if(counts[x] == 1) sum += x;
            // 第 2 次出現：x 不再唯一，把之前加的那次扣掉
            else if(counts[x] == 2) sum -= x;
            // 第 3 次以後：已經扣過了，不需要再處理
        }
        return sum;
    }
};
```

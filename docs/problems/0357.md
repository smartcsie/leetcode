```cpp
/**
 * 題目：357. Count Numbers with Unique Digits
 * 難度：中等 (Medium)
 * 描述：給定一個非負整數 n，計算在 [0, 10^n) 範圍內，所有各位數字都不重複的數字個數。
 *
 * 時間複雜度：O(N)
 * 空間複雜度：O(1)
 *
 * 解法思路：
 * （排列計數）：
 * - 0 位數：只有 0，貢獻 1。
 * - 1 位數：0~9，共 9 個（不含 0 已算在上面）。
 * - 2 位數：首位 9 種（1~9），第二位 9 種（0~9 排除首位），共 9*9。
 * - 3 位數：9 * 9 * 8。
 * - k 位數：9 * 9 * 8 * 7 * ... * (10 - k + 1)。
 *
 * cur 記錄當前位數的不重複數字個數，每輪乘以 (10 - i) 累進：
 * i=1: cur = 9        （1 位數）
 * i=2: cur = 9 * 9    （2 位數）
 * i=3: cur = 9 * 9 * 8（3 位數）
 */
class Solution {
public:
    int countNumbersWithUniqueDigits(int n) {
        int ans = 1;   // 0 本身
        int cur = 9;   // 當前位數的不重複數字個數，從 1 位數的 9 開始
        for (int i = 1; i <= n; i++) {
            ans += cur;
            cur *= 10 - i;  // 下一位的可選數字減少一個
        }
        return ans;
    }
};
```

```cpp
/**
 * 題目：717. 1-bit and 2-bit Characters (1-bit 與 2-bit 字元)
 * 難度：簡單 (Easy)
 * 描述：判斷最後一個字元是否為 1-bit 字元。
 *
 * 時間複雜度：O(N)
 * 空間複雜度：O(1)
 *
 * 解法思路：
 * 1. 簡化指標移動邏輯。
 * 2. 移除不必要的變數以節省空間。
 */

class Solution {
public:
    bool isOneBitCharacter(vector<int>& bits) {
        int n = bits.size();
        int idx = 0;
        while(idx < n - 1) {
            idx += bits[idx] + 1;
        }
        return idx == n - 1;
    }
};
```
